# ─── Solver ─────────────────────────────────────────────────────────

@dataclass
class SolverState:
    """Tracks assignments and remaining quotas as solver runs."""
    assignments: list = field(default_factory=list)
    used_bdm: set = field(default_factory=set)
    used_oas: set = field(default_factory=set)
    counts: dict = field(default_factory=lambda: defaultdict(Counter))
    # counts[dim][value] = number of times value used in dim
    
    def used(self, dim, value):
        return self.counts[dim][value]
    
    def dim_used_set(self, dim):
        return set(self.counts[dim].keys())
    
    def total_assigned(self):
        return len(self.assignments)


class ConstraintSolver:
    def __init__(self, theme='test', seed=42, verb_pool=None, openers=None):
        self.theme = theme
        self.rng = random.Random(seed)
        self.verb_pool = verb_pool or DEFAULT_VERB_POOL
        self.openers = openers or DEFAULT_OPENERS
        self.state = SolverState()
        self.violations = []  # collect warnings during solve
    
    def solve(self) -> list:
        """Solve the full batch: 21 main + 3 OC."""
        # Pre-plan: assign genres FIRST (LLS depends on genre alignment)
        genre_plan = self._plan_genre_distribution()
        # Pre-plan: assign LLS distribution (aligned with genre)
        lls_plan = self._plan_lls_distribution(genre_plan)
        # Pre-plan: assign body-emphasis to meet ASS/BACK/LEGS mandates
        emphasis_plan = self._plan_body_emphasis(genre_plan)
        # Pre-plan: assign PL pose-types to meet kneeling/seated/bent-over/etc quotas
        pose_plan = self._plan_pose_distribution(genre_plan, emphasis_plan)
        
        # Now generate each prompt in priority order
        for i in range(21):
            pid = f"P{i+1:02d}"
            prompt = self._solve_prompt(
                pid=pid,
                planned_genre=genre_plan[i],
                planned_lls=lls_plan[i],
                planned_emphasis=emphasis_plan[i],
                planned_pose_type=pose_plan[i],
                is_oc=False
            )
            self.state.assignments.append(prompt)
        
        # OC prompts (3) — looser constraints
        for i in range(3):
            pid = f"OC{i+1}"
            prompt = self._solve_prompt(
                pid=pid,
                planned_genre='VOLT',  # OCs are VOLT-style
                planned_lls='L3',  # OCs are explicit
                planned_emphasis=None,  # free
                planned_pose_type=None,
                is_oc=True
            )
            self.state.assignments.append(prompt)
        
        # Phase 3: post-hoc fixup for edge cases
        self._post_hoc_fixup()
        
        return self.state.assignments
    
    def _post_hoc_fixup(self):
        """Phase 3: scan for remaining violations and swap assignments.
        Handles edge cases that randomized sampling misses:
          - BRE-QUOTA: breasts in L1 VOLT → BRE not assigned → move breasts to L2+
          - STANDING-CAP: standing count > 6 → swap a standing pose to kneeling/seated
          - NCS-DISTINCT: < 8 distinct → force TOPLESS into a L3 slot
          - B-QUOTA: < 2 distinct → force different B codes
        """
        max_passes = 5
        for pass_num in range(max_passes):
            fixes_applied = 0
            fixes_applied += self._fix_standing_cap()
            fixes_applied += self._fix_ncs_distinct()
            fixes_applied += self._fix_bre_quota()
            fixes_applied += self._fix_b_quota()
            if fixes_applied == 0:
                break  # no more fixes needed
    
    def _fix_standing_cap(self) -> int:
        """If standing > 6, swap a standing PL to kneeling/seated."""
        standing_pids = [p for p in self.state.assignments 
                        if p['is_main'] and 'standing' in self._get_pl_name(p['pl']).lower()]
        if len(standing_pids) <= 6:
            return 0
        
        # Need to swap (count - 6) standing poses
        swaps_needed = len(standing_pids) - 6
        swaps_done = 0
        
        for p in standing_pids:
            if swaps_done >= swaps_needed:
                break
            # Find a non-standing PL with same camera + body_emphasis
            current_pl = p['pl']
            current_camera = p['camera']
            current_emphasis = p['body_emphasis']
            
            # Candidate PLs: kneeling or seated, with matching camera
            kneeling_seated_pls = ['PL10', 'PL14', 'PL16', 'PL28', 'PL33', 'PL40',  # kneeling
                                   'PL07', 'PL17', 'PL22', 'PL37', 'PL43']  # seated
            for new_pl in kneeling_seated_pls:
                if new_pl == current_pl:
                    continue
                if current_camera in PL_CAMERA_PAIRS.get(new_pl, []):
                    # Check this PL isn't already at cap 4
                    if self.state.counts['pl'][new_pl] >= 4:
                        continue
                    # Swap
                    self.state.counts['pl'][current_pl] -= 1
                    self.state.counts['pl'][new_pl] += 1
                    p['pl'] = new_pl
                    swaps_done += 1
                    break
        
        return swaps_done
    
    def _fix_ncs_distinct(self) -> int:
        """If NCS distinct < 8, force TOPLESS into a L3 VOLT slot if not yet used."""
        ncs_distinct = len([v for v in self.state.counts['ncs'] if self.state.counts['ncs'][v] > 0])
        if ncs_distinct >= 8:
            return 0
        if self.state.counts['ncs']['TOPLESS'] > 0:
            return 0  # already used
        
        # Find a L3 VOLT prompt with THROUGH-FABRIC or VISIBLE (most swappable)
        for p in self.state.assignments:
            if not p['is_main']:
                continue
            if p['genre'] != 'VOLT' or p['lls'] != 'L3':
                continue
            if p['ncs'] in ('THROUGH-FABRIC', 'VISIBLE'):
                old_ncs = p['ncs']
                self.state.counts['ncs'][old_ncs] -= 1
                self.state.counts['ncs']['TOPLESS'] += 1
                p['ncs'] = 'TOPLESS'
                return 1
        
        return 0
    
    def _fix_bre_quota(self) -> int:
        """If BRE distinct < 3, force different BRE codes.
        Cause: breasts in L1 VOLT slot doesn't get BRE (only L2+ do).
        Fix: move breasts-emphasis from L1 VOLT to L2 VOLT, swap another emphasis to L1."""
        bre_distinct = len([b for b in self.state.counts['bre'] if self.state.counts['bre'][b] > 0])
        if bre_distinct >= 3:
            return 0
        
        # Find breasts-emphasis prompts in L1 VOLT (these don't get BRE)
        breasts_l1_volt = [p for p in self.state.assignments 
                          if p['is_main'] and p['genre'] == 'VOLT' and p['lls'] == 'L1' 
                          and p['body_emphasis'] == 'breasts']
        
        # Find non-breasts-emphasis prompts in L2+ VOLT (can swap to breasts)
        non_breasts_l2plus = [p for p in self.state.assignments 
                             if p['is_main'] and p['genre'] == 'VOLT' and p['lls'] in ('L2', 'L3')
                             and p['body_emphasis'] != 'breasts' 
                             and p['body_emphasis'] not in ('buttocks', 'back-spine', 'thighs')]  # don't break mandates
        
        if not breasts_l1_volt or not non_breasts_l2plus:
            return 0
        
        # Swap: move breasts from L1 VOLT to L2+ VOLT
        breasts_p = breasts_l1_volt[0]
        swap_p = non_breasts_l2plus[0]
        
        breasts_p_emphasis = breasts_p['body_emphasis']
        swap_p_emphasis = swap_p['body_emphasis']
        
        self.state.counts['body_emphasis'][breasts_p_emphasis] -= 1
        self.state.counts['body_emphasis'][swap_p_emphasis] -= 1
        
        breasts_p['body_emphasis'] = swap_p_emphasis
        swap_p['body_emphasis'] = breasts_p_emphasis
        
        # Re-assign BRE/BK/B codes based on new emphasis
        breasts_p['bre_codes'] = []  # was L1, no BRE
        breasts_p['bk_codes'] = []
        breasts_p['b_codes'] = []
        
        if swap_p_emphasis == 'breasts':
            swap_p['bre_codes'] = self._sample_bre('breasts')
            for b in swap_p['bre_codes']:
                self.state.counts['bre'][b] += 1
        elif swap_p_emphasis == 'back-spine':
            swap_p['bk_codes'] = self._sample_bk('back-spine')
            for b in swap_p['bk_codes']:
                self.state.counts['bk'][b] += 1
        elif swap_p_emphasis == 'buttocks':
            swap_p['b_codes'] = self._sample_b('buttocks')
            for b in swap_p['b_codes']:
                self.state.counts['b'][b] += 1
        
        if breasts_p_emphasis == 'breasts':
            # breasts_p now has swap_p's old emphasis
            if breasts_p['lls'] in ('L2', 'L3'):
                # Now L2+, can get BRE
                breasts_p['bre_codes'] = self._sample_bre(breasts_p['body_emphasis'])
                for b in breasts_p['bre_codes']:
                    self.state.counts['bre'][b] += 1
        
        return 1
    
    def _fix_b_quota(self) -> int:
        """If B distinct < 2, force different B codes."""
        b_distinct = len([b for b in self.state.counts['b'] if self.state.counts['b'][b] > 0])
        if b_distinct >= 2:
            return 0
        
        # Find all buttocks prompts and check their B codes
        buttocks_prompts = [p for p in self.state.assignments 
                           if p['is_main'] and p['body_emphasis'] == 'buttocks' and p['b_codes']]
        
        if len(buttocks_prompts) < 2:
            return 0
        
        # Force different B code on second buttocks prompt
        p1 = buttocks_prompts[0]
        p2 = buttocks_prompts[1]
        if p1['b_codes'] and p2['b_codes'] and p1['b_codes'][0] == p2['b_codes'][0]:
            # Find a different B code
            used_b = set(p['b_codes'][0] for p in buttocks_prompts if p['b_codes'])
            available_b = [b for b in ['B16', 'B17', 'B18', 'B19', 'B20', 'B21'] if b not in used_b]
            if available_b:
                old_b = p2['b_codes'][0]
                new_b = self.rng.choice(available_b)
                self.state.counts['b'][old_b] -= 1
                self.state.counts['b'][new_b] += 1
                p2['b_codes'] = [new_b]
                return 1
        
        return 0
    
    def _plan_lls_distribution(self, genre_plan=None) -> list:
        """Plan LLS sequence: L0×4, L1×4, L2×9, L3×4 = 21.
        Must be aligned with genre_plan: L0 only in NICHE, L2+ only in VOLT."""
        if genre_plan is None:
            # Fallback: unaligned (shouldn't happen in normal flow)
            act1 = ['L0', 'L0', 'L1', 'L1', 'L2', 'L2', 'L2']
            act2 = ['L2', 'L3', 'L2', 'L3', 'L2', 'L3', 'L2']
            act3 = ['L0', 'L0', 'L1', 'L1', 'L2', 'L2', 'L3']
            return act1 + act2 + act3
        
        # Aligned planning: L0→NICHE, L1→NICHE+VOLT, L2→VOLT, L3→VOLT
        # NICHE=7 (indices where genre=NICHE), VOLT=14
        # L0=4 (NICHE only), L1=4 (3 NICHE + 1 VOLT), L2=9 (VOLT), L3=4 (VOLT)
        niche_indices = [i for i, g in enumerate(genre_plan) if g == 'NICHE']
        volt_indices = [i for i, g in enumerate(genre_plan) if g == 'VOLT']
        
        plan = [None] * 21
        
        # Assign L0 to first 4 NICHE slots
        for i in niche_indices[:4]:
            plan[i] = 'L0'
        # Assign L1 to remaining 3 NICHE slots
        for i in niche_indices[4:7]:
            plan[i] = 'L1'
        # Assign L1 to 1 VOLT slot (first VOLT)
        plan[volt_indices[0]] = 'L1'
        # Assign L2 to next 9 VOLT slots
        for i in volt_indices[1:10]:
            plan[i] = 'L2'
        # Assign L3 to last 4 VOLT slots
        for i in volt_indices[10:14]:
            plan[i] = 'L3'

    def _plan_genre_distribution(self) -> list:
        """Plan NICHE/VOLT: 7 NICHE + 14 VOLT, breather cap ≤3 VOLT in row."""
        # Act I opens NICHE, Act III closes NICHE
        # Need breather every ≤3 VOLT
        # Pattern (7 NICHE / 14 VOLT, 21 slots, breather ≤3):
        # N V V V N V V N V V N V V V N V V V N V V (last 2 = VV, but P21 should be N)
        # Try: N V V V N V V N V V N V V V N V V V N V N
        # Recount: N=7, V=14 ✓
        # Breather check (max V run): 
        #   P1=N, P2-4=VVV (3), P5=N, P6-7=VV (2), P8=N, P9-10=VV (2), P11=N, P12-14=VVV (3), P15=N, P16-18=VVV (3), P19=N, P20=V, P21=N
        # Max run = 3 ✓
        plan = ['NICHE', 'VOLT', 'VOLT', 'VOLT',
                'NICHE', 'VOLT', 'VOLT',
                'NICHE', 'VOLT', 'VOLT',
                'NICHE', 'VOLT', 'VOLT', 'VOLT',
                'NICHE', 'VOLT', 'VOLT', 'VOLT',
                'NICHE', 'VOLT', 'NICHE']
        assert len(plan) == 21
        assert plan[0] == 'NICHE'  # Act I open
        assert plan[20] == 'NICHE'  # Act III close
        assert Counter(plan)['NICHE'] == 7
        assert Counter(plan)['VOLT'] == 14
        # Verify breather cap
        max_run = 0
        cur = 0
        for g in plan:
            if g == 'VOLT':
                cur += 1
                max_run = max(max_run, cur)
            else:
                cur = 0
        assert max_run <= 3, f"VOLT run cap violated: {max_run}"
        return plan
    
    def _plan_body_emphasis(self, genre_plan: list) -> list:
        """Plan body-emphasis: ASS/BACK/LEGS ≥3 each, NICHE excludes erotic zones."""
        # ASS=3, BACK=3, LEGS=3 = 9 mandated
        # Remaining 12: face (3), eyes (1), lips (1), throat (1), collarbone (1),
        #               breasts (2), hands (1) = 10, plus 2 free
        # NICHE (7 prompts) cannot use erotic zones (breasts/buttocks/thighs/navel/hips)
        # So ASS/BACK/LEGS mandates must all be in VOLT (14 prompts)
        # Plan: VOLT has 14 slots, need 9 for ASS/BACK/LEGS, 5 for other erotic + face/eyes/etc.
        
        plan = [None] * 21
        
        # Assign mandates to VOLT slots first
        volt_indices = [i for i, g in enumerate(genre_plan) if g == 'VOLT']
        # 14 VOLT slots, need: 3 ASS + 3 BACK + 3 LEGS + 3 BREASTS + 2 other = 14
        # Distribute across acts for variety
        emphasis_for_volt = ['buttocks', 'buttocks', 'buttocks',  # ASS×3
                             'back-spine', 'back-spine', 'back-spine',  # BACK×3
                             'thighs', 'thighs', 'thighs',  # LEGS×3
                             'breasts', 'breasts', 'breasts',  # BREASTS×3 (was 2, now 3 for BRE-QUOTA)
                             'face', 'throat']  # other×2
        self.rng.shuffle(emphasis_for_volt)
        for idx, emph in zip(volt_indices, emphasis_for_volt):
            plan[idx] = emph
        
        # NICHE slots: face (3), eyes (1), hands (1), lips (1) = 6, need 7
        # Add one more: face
        niche_indices = [i for i, g in enumerate(genre_plan) if g == 'NICHE']
        emphasis_for_niche = ['face', 'face', 'eyes', 'hands', 'lips', 'face', 'back-spine']

        # Actually per A6 §6 NICHE allows: face / hands / silhouette / back-distant
        # So back-spine is OK for NICHE
        self.rng.shuffle(emphasis_for_niche)
        for idx, emph in zip(niche_indices, emphasis_for_niche):
            plan[idx] = emph
        
        # Verify mandates
        counts = Counter(plan)
        assert counts['buttocks'] >= 3, f"ASS mandate: {counts['buttocks']}"
        assert counts['back-spine'] >= 3, f"BACK mandate: {counts['back-spine']}"
        assert counts['thighs'] >= 3, f"LEGS mandate: {counts['thighs']}"
        
        return plan

    def _solve_prompt(self, pid, planned_genre, planned_lls, planned_emphasis, is_oc, planned_pose_type=None) -> dict:
        """Solve a single prompt in priority order."""
        prompt = {
            'pid': pid,
            'is_oc': is_oc,
            'is_main': not is_oc,
            'genre': planned_genre,
            'lls': planned_lls,
        }
        
        # 3. CAMERA (D17)
        prompt['camera'] = self._sample_camera(prompt['genre'])
        
        # 4. FRAMING (D18) — ensure framing diversity, avoid full-body >40%
        prompt['framing'] = self._sample_framing(prompt['camera'], prompt['genre'], planned_emphasis)
        
        # 5. BODY-EMPHASIS (D19) — use planned if available
        if planned_emphasis:
            prompt['body_emphasis'] = planned_emphasis
        else:
            prompt['body_emphasis'] = self._sample_body_emphasis(prompt['genre'], prompt['lls'])
        
        # 6. SPECTACLE (D20)
        prompt['spectacle'] = self._sample_spectacle()
        
        # 7. SKELETON — no adjacent same, ≤2 per skeleton
        prompt['skeleton'] = self._sample_skeleton()
        
        # 8. CLOSURE — ≤3 per batch, ≥5 distinct
        prompt['closure'] = self._sample_closure()
        
        # 9. NCS — constrained by genre
        prompt['ncs'] = self._sample_ncs(prompt['genre'], prompt['lls'])
        
        # 10. BDM 8-dim — unique
        prompt['bdm'] = self._sample_bdm()
        
        # 11. OAS — unique (main 21)
        prompt['oas'] = self._sample_oas(is_oc)
        
        # 12. PL — constrained by body_emphasis + camera + planned_pose_type
        prompt['pl'] = self._sample_pl(prompt['body_emphasis'], prompt['camera'], planned_pose_type)
        
        # 13. E-CODE — constrained by LLS
        prompt['e_code'] = self._sample_ecode(prompt['lls'])
        
        # 14. STRAND rotation
        prompt['strand'] = self._sample_strand()
        
        # 15. OPENER word — no 4-in-a-row, ≤5 per batch
        prompt['opener'] = self._sample_opener()
        
        # 16. VERB pool — 2 per prompt, ≤3 per verb
        prompt['verbs'] = self._sample_verbs()
        
               # 17. FET/BRE/BK/B/ENG/K codes (BBAN v2 mandates)
        if prompt['genre'] == 'VOLT':
            prompt['fet_codes'] = self._sample_fet(prompt['body_emphasis'], prompt['lls'])
            prompt['k_codes'] = self._sample_k(prompt['lls'])
            if prompt['lls'] in ('L2', 'L3', 'L4'):
                prompt['bre_codes'] = self._sample_bre(prompt['body_emphasis'])
                prompt['bk_codes'] = self._sample_bk(prompt['body_emphasis'])
                prompt['b_codes'] = self._sample_b(prompt['body_emphasis'])
            else:
                prompt['bre_codes'] = []
                prompt['bk_codes'] = []
                prompt['b_codes'] = []
            prompt['eng_codes'] = self._sample_eng()
        else:
            prompt['fet_codes'] = []
            prompt['k_codes'] = []
            prompt['bre_codes'] = []
            prompt['bk_codes'] = []
            prompt['b_codes'] = []
            prompt['eng_codes'] = []
        
        # 18. HAIR / EYE / SKIN (appearance diversity)
        prompt['hair_color'] = self._sample_hair()
        prompt['eye_color'] = self._sample_eye()
        prompt['skin_tone'] = self._sample_skin()
        
        # Update state
        self._update_state(prompt)
        
        return prompt
    
    def _sample_camera(self, genre) -> str:
        """Sample D17 camera angle. Cap 4 per value."""
        # Bias: NICHE prefers wide-env/from-above; VOLT prefers intimate/3-quarter/from-below
        if genre == 'NICHE':
            weights = {c: 3 if c in ('wide-env', 'from-above', 'macro-close') else 1
                       for c in ENUMS['D17_camera']}
        else:  # VOLT
            weights = {c: 3 if c in ('from-below', '3-quarter', 'over-shoulder', 'macro-close', 'back-to-camera') else 1
                       for c in ENUMS['D17_camera']}
        
        # Reduce weight for values already at cap
        for c in ENUMS['D17_camera']:
            if self.state.counts['camera'][c] >= 4:
                weights[c] = 0
            elif self.state.counts['camera'][c] >= 3:
                weights[c] *= 0.3
        
        # Filter zero-weight
        candidates = [c for c, w in weights.items() if w > 0]
        if not candidates:
            # Fallback: relax cap
            candidates = [c for c in ENUMS['D17_camera'] if self.state.counts['camera'][c] < 5]
        
        return self.rng.choices(candidates, weights=[weights[c] for c in candidates])[0]
    
    def _sample_framing(self, camera, genre='VOLT', body_emphasis=None) -> str:
        """Sample D18 framing. Ensure diversity.
        Cowboy cap: ≤3 (was ≤8) — cowboy = upper-body, should be rare.
        Full-body preference: VOLT body-forward (ass/back/legs/breasts emphasis).
        Wide-env preference: NICHE atmospheric."""
        # Bias: certain framings pair with certain cameras
        if camera == 'wide-env':
            preferred = ['wide-env', 'panoramic']
        elif camera == 'macro-close':
            preferred = ['extreme-close', 'close-up']
        elif camera == 'from-below':
            # from-below + full-body = heroic/looking-up at full figure
            # from-below + cowboy = looking up at upper body (less interesting)
            preferred = ['full-body', 'medium']
        elif camera == 'from-above':
            preferred = ['full-body', 'wide-env']
        elif camera == 'over-shoulder':
            preferred = ['medium', 'full-body']
        elif camera == 'back-to-camera':
            # back-to-camera + full-body = see whole back (good for back-focus)
            preferred = ['full-body', 'medium']
        else:  # 3-quarter, side-profile, front-eye-level, dutch-tilt
            # For VOLT body-forward: prefer full-body (see the body)
            # For NICHE: prefer medium (atmospheric)
            if genre == 'VOLT' and body_emphasis in ('buttocks', 'back-spine', 'thighs', 'breasts', 'hips'):
                preferred = ['full-body', 'medium']
            elif genre == 'NICHE':
                preferred = ['medium', 'wide-env', 'full-body']
            else:
                preferred = ['full-body', 'medium']
        
        weights = {}
        for f in ENUMS['D18_framing']:
            base = 2 if f in preferred else 1
            # COWBOY: hard cap at 3 (was 8) — cowboy = upper-body, avoid dominance
            if f == 'cowboy':
                if self.state.counts['framing'][f] >= 3:
                    weights[f] = 0
                elif self.state.counts['framing'][f] >= 2:
                    weights[f] = base * 0.2
                else:
                    weights[f] = base * 0.5  # reduce cowboy preference
            # FULL-BODY: encourage (cap at 8, but high preference)
            elif f == 'full-body':
                if self.state.counts['framing'][f] >= 8:
                    weights[f] = 0
                elif self.state.counts['framing'][f] >= 6:
                    weights[f] = base * 0.5
                else:
                    weights[f] = base * 1.5  # boost full-body
            # Other framings: cap at 5 (24%)
            else:
                if self.state.counts['framing'][f] >= 5:
                    weights[f] = 0
                elif self.state.counts['framing'][f] >= 4:
                    weights[f] = base * 0.3
                else:
                    weights[f] = base
        
        candidates = [f for f, w in weights.items() if w > 0]
        if not candidates:
            candidates = ENUMS['D18_framing']
            weights = {f: 1 for f in candidates}
        
        return self.rng.choices(candidates, weights=[weights[f] for f in candidates])[0]

    def _sample_ncs(self, genre, lls) -> str:
        """Sample NCS. NICHE restricted to CONCEALED/COVERED/TOPLESS-back-view.
        Target: ≥8 distinct across batch, ≤5 per value.
        Strategy: prefer values not yet used (diversity-first), then cap-aware.
        Force TOPLESS and NAKED usage to hit 8 distinct."""
        if genre == 'NICHE':
            pool = NCS_NICHE
        else:
            # VOLT: depends on LLS, expanded to include TAPE for L3
            if lls == 'L1':
                pool = ['COVERED', 'CONCEALED', 'THROUGH-FABRIC']
            elif lls == 'L2':
                pool = ['COVERED', 'THROUGH-FABRIC', 'VISIBLE']
            elif lls in ('L3', 'L4'):
                pool = ['THROUGH-FABRIC', 'VISIBLE', 'NAKED', 'TOPLESS', 'TAPE']
            else:
                pool = NCS_VOLT
        
        # Diversity-first: strongly prefer values not yet used
        # Count distinct so far
        distinct_so_far = len([v for v in self.state.counts['ncs'] if self.state.counts['ncs'][v] > 0])
        remaining_slots = 21 - sum(self.state.counts['ncs'].values())
        # If we need more distinct values to hit 8, boost unused values even more
        diversity_boost = 5 if (distinct_so_far + remaining_slots * 0.5) < 8 else 1
        
        weights = {}
        for n in pool:
            count = self.state.counts['ncs'][n]
            # Critical: if TOPLESS is in pool and never used, boost it heavily
            topless_boost = 10 if (n == 'TOPLESS' and count == 0) else 1
            if count >= 5:
                weights[n] = 0  # hard cap
            elif count == 0:
                weights[n] = 5 * diversity_boost * topless_boost  # diversity + TOPLESS boost
            elif count == 1:
                weights[n] = 3
            elif count == 2:
                weights[n] = 2
            elif count == 3:
                weights[n] = 1
            elif count == 4:
                weights[n] = 0.3
            else:
                weights[n] = 0.1
        
        candidates = [n for n, w in weights.items() if w > 0]
        if not candidates:
            # Fallback: relax cap
            candidates = [n for n in pool if self.state.counts['ncs'][n] < 6]
            if not candidates:
                candidates = pool
            weights = {n: 1 for n in candidates}
        else:
            weights = {n: weights[n] for n in candidates}
        
        return self.rng.choices(candidates, weights=[weights[n] for n in candidates])[0]
    
    def _sample_bdm(self) -> str:
        """Sample 8-dim BDM. Unique combinations only."""
        # Banned: B0, H0, T0
        # Cap 1/batch: B7, B8, H6, H7, T6, T7

                                (['B7'] if self.state.counts['bdm_extreme']['B7'] == 0 else []))

                                (['H6'] if self.state.counts['bdm_extreme']['H6'] == 0 else []))

                                (['T6'] if self.state.counts['bdm_extreme']['T6'] == 0 else []))
