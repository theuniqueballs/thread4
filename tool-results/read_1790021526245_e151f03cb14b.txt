    32→function matches(item: unknown, q: string): boolean {
    33→  if (q === '') return true
    34→  try {
    35→    return JSON.stringify(item).toLowerCase().includes(q)
    36→  } catch {
    37→    return true
    38→  }
    39→}
    40→
    41→function firstArray(spec: unknown, keys: string[]): unknown[] | null {
    42→  const rec = asRecord(spec)
    43→  for (const k of keys) {
    44→    if (Array.isArray(rec[k])) return rec[k]
    45→  }
    46→  if (Array.isArray(spec)) return spec
    47→  return null
    48→}
    49→
    50→/** Records from an array of objects or a keyed object of objects. */
    51→function recordItems(v: unknown): { key: string; rec: Record<string, unknown> }[] {
    52→  if (Array.isArray(v)) {
    53→    return v
    54→      .map((x, i) => ({ key: String(i), rec: asRecord(x) }))
    55→      .filter((x) => Object.keys(x.rec).length > 0 || typeof v[x.rec ? 0 : 0] === 'object')
    56→  }
    57→  return Object.entries(asRecord(v)).map(([k, val]) => ({ key: k, rec: asRecord(val) }))
    58→}

    60→function normalizeDegree(v: unknown): string {
    61→  const s = asStr(v).trim().toUpperCase()
    62→  if (s === 'LOW') return 'L'
    63→  if (s === 'MEDIUM' || s === 'MED') return 'M'
    64→  if (s === 'HIGH') return 'H'
    65→  if (s === 'L' || s === 'M' || s === 'H') return s
    66→  return ''
    67→}

    75→function CarriersView({ spec, filter }: { spec: unknown; filter: string }) {
    76→  type Carrier = { name: string; cls: string; degree: string; mechanism: string; note: string; raw: unknown }
    77→  const carriers: Carrier[] = []
    78→
    79→  const push = (rec: Record<string, unknown>, raw: unknown, fallbackCls = '') => {
    80→    const name = pickStr(rec, 'name', 'id', 'label', 'carrier', 'tag')
    81→    if (!name) return
    82→    carriers.push({
    83→      name,
    84→      cls: pickStr(rec, 'class', 'cls', 'code', 'carrier_class') || fallbackCls,
    85→      degree: normalizeDegree(rec.degree ?? rec.deg ?? rec.level),
    86→      mechanism: pickStr(rec, 'mechanism', 'mechanism_group', 'group', 'family', 'type'),
    87→      note: pickStr(rec, 'note', 'description', 'desc', 'why', 'use', 'usage'),
    88→      raw,
    89→    })
    90→  }
    91→
    92→  const root = asRecord(spec)
    93→  const direct = firstArray(spec, ['carriers', 'items'])
    94→  if (direct) {
    95→    for (const item of direct) {
    96→      const rec = asRecord(item)
    97→      if (Object.keys(rec).length === 0) {
    98→        const s = asStr(item)
    99→        if (s) carriers.push({ name: s, cls: '', degree: '', mechanism: '', note: '', raw: item })
   100→      } else push(rec, item)
   101→    }
   102→  }
   103→  // shape: { classes: [{ code, name, carriers: [...] }] }
   104→  const classGroups = asArray(root.classes)
   105→  if (classGroups.length > 0) {
   106→    for (const g of classGroups) {
   107→      const gRec = asRecord(g)
   108→      const cls = pickStr(gRec, 'code', 'class', 'cls', 'id', 'name')
   109→      for (const item of asArray(gRec.carriers ?? gRec.items)) {
   110→        push(asRecord(item), item, cls)
   111→      }
   112→    }
   113→  }
   114→  // shape: keyed object of class → array
   115→  if (carriers.length === 0) {
   116→    for (const [k, v] of Object.entries(root)) {
   117→      if (Array.isArray(v)) {
   118→        for (const item of v) push(asRecord(item), item, k)
   119→      }
   120→    }
   121→  }
   122→
   123→  if (carriers.length === 0) return <GenericSpecView data={spec} filter={filter} />
   124→
   125→  const filtered = carriers.filter((c) => matches(c.raw, filter))
   126→  if (filtered.length === 0) return <FilterEmpty filter={filter} />
   127→
   128→  const groups = new Map<string, Carrier[]>()
   129→  for (const c of filtered) {
   130→    const key = c.cls || '—'
   131→    const list = groups.get(key)
   132→    if (list) list.push(c)
   133→    else groups.set(key, [c])
   134→  }
   135→
   136→  return (
   137→    <div className="t4-scroll max-h-96 space-y-5 overflow-y-auto pr-1">
   138→      {[...groups.entries()].map(([cls, items]) => (
   139→        <div key={cls}>
   140→          <div className="mb-2 flex items-center gap-2">
   141→            <span className="inline-flex size-6 items-center justify-center rounded-md border border-amber-500/40 bg-amber-500/10 font-mono text-xs font-semibold text-amber-400">
   142→              {cls}
   143→            </span>
   144→            <span className="text-xs font-medium uppercase tracking-wide text-zinc-500">
   145→              {cls === '—' ? 'носители' : `класс ${cls}`} · {items.length}
   146→            </span>
   147→          </div>
   148→          <div className="overflow-hidden rounded-lg border border-zinc-800">
   149→            <Table>
   150→              <TableHeader>
   151→                <TableRow className="hover:bg-transparent">
   152→                  <TableHead className="text-zinc-400">Носитель</TableHead>
   153→                  <TableHead className="text-zinc-400">Степень</TableHead>
   154→                  <TableHead className="text-zinc-400">Механизм</TableHead>
   155→                  <TableHead className="text-zinc-400">Примечание</TableHead>
   156→                </TableRow>
   157→              </TableHeader>
   158→              <TableBody>
   159→                {items.map((c, i) => (
   160→                  <TableRow key={`${c.name}-${i}`} className="border-zinc-800/70">
   161→                    <TableCell className="font-medium text-zinc-200">{c.name}</TableCell>
   162→                    <TableCell>{c.degree ? <DegreeBadge degree={c.degree} /> : <span className="text-zinc-600">—</span>}</TableCell>
   163→                    <TableCell className="text-zinc-400">{c.mechanism || '—'}</TableCell>
   164→                    <TableCell className="max-w-xs whitespace-normal text-xs leading-relaxed text-zinc-500">{c.note || '—'}</TableCell>
   165→                  </TableRow>
   166→                ))}
   167→              </TableBody>
   168→            </Table>
   169→          </div>
   170→        </div>
   171→      ))}
   172→    </div>
   173→  )
   174→}

   180→function riskTone(risk: string): string {
   181→  const r = risk.toLowerCase()
   182→  if (['safe', 'low', 'clean', 'ok'].includes(r)) return 'border-emerald-600/40 bg-emerald-600/10 text-emerald-400'
   183→  if (['edge', 'risky', 'medium', 'med', 'warn', 'spicy'].includes(r)) return 'border-amber-500/40 bg-amber-500/10 text-amber-400'
   184→  if (['banned', 'high', 'hard', 'danger'].includes(r)) return 'border-rose-500/40 bg-rose-500/10 text-rose-400'
   185→  return 'border-zinc-700 bg-zinc-800/60 text-zinc-300'
   186→}

   188→function PosesView({ spec, filter }: { spec: unknown; filter: string }) {
   189→  type Pose = { name: string; category: string; risk: string; note: string; raw: unknown }
   190→  const poses: Pose[] = []
   191→  const push = (item: unknown, fallbackCat = '') => {
   192→    const rec = asRecord(item)
   193→    const name = pickStr(rec, 'name', 'id', 'pose', 'tag') || asStr(item)
   194→    if (!name || (Object.keys(rec).length === 0 && typeof item !== 'string')) return
   195→    poses.push({
   196→      name,
   197→      category: pickStr(rec, 'category', 'type', 'family', 'group', 'class') || fallbackCat,
   198→      risk: pickStr(rec, 'risk', 'risk_level', 'edge', 'safety', 'level'),
   199→      note: pickStr(rec, 'note', 'description', 'desc', 'why', 'use'),
   200→      raw: item,
   201→    })
   202→  }
   203→  const direct = firstArray(spec, ['poses', 'items'])
   204→  if (direct) for (const item of direct) push(item)
   205→  const root = asRecord(spec)
   206→  if (poses.length === 0) {
   207→    for (const [k, v] of Object.entries(root)) {
   208→      if (Array.isArray(v)) for (const item of v) push(item, k)
   209→    }
   210→  }
   211→  if (poses.length === 0) return <GenericSpecView data={spec} filter={filter} />
   212→
   213→  const filtered = poses.filter((p) => matches(p.raw, filter))
   214→  if (filtered.length === 0) return <FilterEmpty filter={filter} />
   215→
   216→  return (
   217→    <div className="t4-scroll max-h-96 overflow-y-auto pr-1">
   218→      <div className="overflow-hidden rounded-lg border border-zinc-800">
   219→        <Table>
   220→          <TableHeader>
   221→            <TableRow className="hover:bg-transparent">
   222→              <TableHead className="text-zinc-400">Поза</TableHead>
   223→              <TableHead className="text-zinc-400">Категория</TableHead>
   224→              <TableHead className="text-zinc-400">Риск</TableHead>
   225→              <TableHead className="text-zinc-400">Примечание</TableHead>
   226→            </TableRow>
   227→          </TableHeader>
   228→          <TableBody>
   229→            {filtered.map((p, i) => (
   230→              <TableRow key={`${p.name}-${i}`} className="border-zinc-800/70">
   231→                <TableCell className="font-medium text-zinc-200">{p.name}</TableCell>
   232→                <TableCell className="text-zinc-400">{p.category || '—'}</TableCell>
   233→                <TableCell>
   234→                  {p.risk ? (
   235→                    <span className={cn('inline-flex items-center rounded-md border px-1.5 py-0.5 text-[11px] font-medium', riskTone(p.risk))}>
   236→                      {p.risk}
   237→                    </span>
   238→                  ) : (
   239→                    <span className="text-zinc-600">—</span>
   240→                  )}
   241→                </TableCell>
   242→                <TableCell className="max-w-xs whitespace-normal text-xs leading-relaxed text-zinc-500">{p.note || '—'}</TableCell>
   243→              </TableRow>
   244→            ))}
   245→          </TableBody>
   246→        </Table>
   247→      </div>
   248→    </div>
   249→  )
   250→}

   279→function hexOf(name: string, explicit?: string): string | null {
   280→  const check = (s: string): string | null => {
   281→    const t = s.trim()
   282→    if (/^#?[0-9a-f]{3}$/.test(t)) return `#${t.replace('#', '')}`
   283→    if (/^#?[0-9a-f]{6}$/.test(t)) return t.startsWith('#') ? t : `#${t}`
   284→    return null
   285→  }
   286→  if (explicit) {
   287→    const h = check(explicit)
   288→    if (h) return h
   289→  }
   290→  const key = name.trim().toLowerCase().replace(/[_-]+/g, ' ')
   291→  return COLOR_HEX[key] ?? null
   292→}

   294→function PalettesView({ spec, filter }: { spec: unknown; filter: string }) {
   295→  type Swatch = { name: string; hex: string | null }
   296→  type Palette = { name: string; note: string; colors: Swatch[]; raw: unknown }
   297→
   298→  const toSwatches = (v: unknown): Swatch[] =>
   299→    asArray(v)
   300→      .map((c): Swatch | null => {
   301→        if (typeof c === 'string') {
   302→          const h = hexOf(c)
   303→          return h || c.match(/^[a-z]/i) ? { name: c, hex: h } : null
   304→        }
   305→        const rec = asRecord(c)
   306→        const name = pickStr(rec, 'name', 'label', 'color', 'title')
   307→        const explicit = pickStr(rec, 'hex', 'value', 'color')
   308→        if (!name && !explicit) return null
   309→        const finalName = name || explicit
   310→        return { name: finalName, hex: hexOf(finalName, explicit) }
   311→      })
   312→      .filter((s): s is Swatch => s != null)
   313→
   314→  // keyed object { colorName: hex } → swatches
   315→  const recSwatches = (rec: Record<string, unknown>): Swatch[] =>
   316→    Object.entries(rec)
   317→      .filter(([, v]) => typeof v === 'string' && (hexOf(v) || hexOf('', v) || /^#?[0-9a-f]{3,6}$/i.test(v)))
   318→      .map(([k, v]) => ({ name: k, hex: hexOf(k, asStr(v)) }))
   319→
   320→  const palettes: Palette[] = []
   321→  const push = (item: unknown) => {
   322→    const rec = asRecord(item)
   323→    const name = pickStr(rec, 'name', 'id', 'palette', 'title')
   324→    const colorVals = ['colors', 'swatches', 'palette', 'shades'].map((k) => rec[k]).find((v) => Array.isArray(v) || asRecord(v).constructor === Object)
   325→    let colors: Swatch[] = []
   326→    if (colorVals != null) colors = Array.isArray(colorVals) ? toSwatches(colorVals) : recSwatches(asRecord(colorVals))
   327→    else colors = recSwatches(rec)
   328→    if (!name && colors.length === 0) return
   329→    palettes.push({
   330→      name: name || 'Без имени',
   331→      note: pickStr(rec, 'note', 'vibe', 'mood', 'description', 'use'),
   332→      colors,
   333→      raw: item,
   334→    })
   335→  }
   336→
   337→  const direct = firstArray(spec, ['palettes', 'items'])
   338→  if (direct) for (const item of direct) push(item)
   339→  if (palettes.length === 0) {
   340→    for (const v of Object.values(asRecord(spec))) {
   341→      if (Array.isArray(v)) for (const item of v) push(item)
   342→    }
   343→  }
   344→  if (palettes.length === 0 && asRecord(spec).constructor === Object && !Array.isArray(spec)) {
   345→    // maybe the spec itself is one palette
   346→    const rec = asRecord(spec)
   347→    if (pickStr(rec, 'name', 'id') || Object.keys(rec).length > 0) push(spec)
   348→  }
   349→
   350→  if (palettes.length === 0 || palettes.every((p) => p.colors.length === 0 && !p.name)) {
   351→    return <GenericSpecView data={spec} filter={filter} />
   352→  }
   353→
   354→  const filtered = palettes.filter((p) => matches(p.raw, filter))
   355→  if (filtered.length === 0) return <FilterEmpty filter={filter} />
   356→
   357→  return (
   358→    <div className="t4-scroll max-h-96 space-y-4 overflow-y-auto pr-1">
   359→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   360→        {filtered.map((p, i) => (
   361→          <div key={`${p.name}-${i}`} className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-4">
   362→            <div className="mb-1 text-sm font-semibold text-zinc-100">{p.name}</div>
   363→            {p.note ? <div className="mb-3 text-xs leading-relaxed text-zinc-500">{p.note}</div> : null}
   364→            <div className="flex flex-wrap gap-1.5">
   365→              {p.colors.length === 0 ? <span className="text-xs text-zinc-600">нет данных</span> : null}
   366→              {p.colors.map((c, j) => (
   367→                <span
   368→                  key={`${c.name}-${j}`}
   369→                  className="inline-flex items-center gap-1.5 rounded-md border border-zinc-700/70 bg-zinc-800/50 py-0.5 pl-0.5 pr-1.5"
   370→                  title={c.hex ?? c.name}
   371→                >
   372→                  <span
   373→                    className="size-4 rounded-[4px] border border-zinc-600/60"
   374→                    style={c.hex ? { backgroundColor: c.hex } : { background: 'linear-gradient(135deg,#3f3f46 0%,#f59e0b 100%)' }}
   375→                  />
   376→                  <span className="text-[11px] text-zinc-300">{c.name}</span>
   377→                </span>
   378→              ))}
   379→            </div>
   380→          </div>
   381→        ))}
   382→      </div>
   383→    </div>
   384→  )
   385→}

   403→function EnginesView({ spec, filter }: { spec: unknown; filter: string }) {
   404→  const engines = recordItems(asRecord(spec).engines ?? spec)
   405→  const usable = engines.filter((e) => Object.keys(e.rec).length > 0)
   406→  if (usable.length === 0) return <GenericSpecView data={spec} filter={filter} />
   407→
   408→  const filtered = usable.filter((e) => matches(e.rec, filter))
   409→  if (filtered.length === 0) return <FilterEmpty filter={filter} />
   410→
   411→  return (
   412→    <div className="t4-scroll max-h-96 space-y-4 overflow-y-auto pr-1">
   413→      <div className="grid gap-4 lg:grid-cols-2">
   414→        {filtered.map((e) => {
   415→          const name = pickStr(e.rec, 'name', 'id', 'title') || humanizeKey(e.key)
   416→          const version = pickStr(e.rec, 'version', 'generation', 'gen')
   417→          const law = pickStr(e.rec, 'law', 'law_summary', 'summary', 'description', 'rule', 'thesis')
   418→          const onBody = pickStr(e.rec, 'on_body', 'body', 'notes', 'usage', 'use')
   419→          const failures = pickStrArray(e.rec, 'failure_modes', 'failures', 'failure', 'anti_patterns', 'traps')
   420→          const { name: _n, version: _v, law: _l, on_body: _o, failure_modes: _f, failures: _ff, ...rest } = e.rec
   421→          const extra = Object.entries(rest).slice(0, 3)
   422→          return (
   423→            <div key={e.key} className="rounded-lg border border-zinc-800 bg-zinc-900/60 p-4">
   424→              <div className="flex items-center justify-between gap-2">
   425→                <div className="text-sm font-semibold text-amber-200">{name}</div>
   426→                {version ? <Chip>{version}</Chip> : null}
   427→              </div>
   428→              {law ? <LabeledBlock label="law">{law}</LabeledBlock> : null}
   429→              {onBody ? <LabeledBlock label="on body">{onBody}</LabeledBlock> : null}
   430→              {failures.length > 0 ? (
   431→                <LabeledBlock label="failure modes" tone="rose">
   432→                  <div className="flex flex-wrap gap-1.5">
   433→                    {failures.map((f, i) => (
   434→                      <Chip key={i} tone="rose">{f}</Chip>
   435→                    ))}
   436→                  </div>
   437→                </LabeledBlock>
   438→              ) : null}
   439→              {extra.map(([k, v]) => {
   440→                const s = asStr(v)
   441→                if (!s) return null
   442→                return (
   443→                  <LabeledBlock key={k} label={humanizeKey(k)}>{s}</LabeledBlock>
   444→                )
   445→              })}
   446→            </div>
   447→          )

   910→
   911→  if (Array.isArray(value)) {
   912→    if (value.length === 0) return <span className="text-xs text-zinc-600">пусто</span>
   913→    if (value.every((x) => typeof x !== 'object' || x === null)) {
   914→      return (
   915→        <span className="flex flex-wrap gap-1">
   916→          {value.map((x, i) => <Chip key={i}>{asStr(x)}</Chip>)}
   917→        </span>
   918→      )
   919→    }
   920→    return <GenericTable items={value} depth={depth} />
   921→  }
   922→
   923→  const rec = asRecord(value)
   924→  const entries = Object.entries(rec)
   925→  if (entries.length === 0) return <span className="text-xs text-zinc-600">пусто</span>
   926→  if (depth >= 3) {
   927→    return <span className="font-mono text-[11px] text-zinc-500">{JSON.stringify(value)}</span>
   928→  }
   929→  return (
   930→    <div className="space-y-2">
   931→      {entries.map(([k, v]) => (
   932→        <div key={k} className={cn(depth > 0 && 'border-l border-zinc-800 pl-3')}>
   933→          <div className="text-[10px] font-semibold uppercase tracking-widest text-zinc-600">{humanizeKey(k)}</div>
   934→          <div className="mt-0.5 text-xs leading-relaxed">
   935→            <GenericValue value={v} depth={depth + 1} />
   936→          </div>
   937→        </div>
   938→      ))}
   939→    </div>
   940→  )
   941→}
   942→
   943→function GenericTable({ items, depth }: { items: unknown[]; depth: number }) {
   944→  const cols: string[] = []
   945→  for (const item of items.slice(0, 20)) {
   946→    for (const k of Object.keys(asRecord(item))) {
   947→      if (!cols.includes(k) && cols.length < 7) cols.push(k)
   948→    }
   949→  }
   950→  if (cols.length === 0) {
   951→    return (
   952→      <div className="space-y-1">
   953→        {items.map((x, i) => <GenericValue key={i} value={x} depth={depth + 1} />)}
   954→      </div>
   955→    )
   956→  }
   957→  return (
   958→    <div className="overflow-hidden rounded-lg border border-zinc-800">
   959→      <Table>
   960→        <TableHeader>
   961→          <TableRow className="hover:bg-transparent">
   962→            {cols.map((c) => (
   963→              <TableHead key={c} className="text-zinc-400">{humanizeKey(c)}</TableHead>
   964→            ))}
   965→          </TableRow>
   966→        </TableHeader>
   967→        <TableBody>
   968→          {items.map((item, i) => (
   969→            <TableRow key={i} className="border-zinc-800/70">
   970→              {cols.map((c) => (
   971→                <TableCell key={c} className="max-w-xs whitespace-normal text-xs text-zinc-400">
   972→                  <GenericValue value={asRecord(item)[c]} depth={depth + 1} />
   973→                </TableCell>
   974→              ))}
   975→            </TableRow>
   976→          ))}
   977→        </TableBody>
   978→      </Table>
   979→    </div>
   980→  )
   981→}
   982→
   983→function GenericSpecView({ data, filter }: { data: unknown; filter: string }) {
   984→  if (Array.isArray(data)) {
   985→    const filtered = data.filter((x) => matches(x, filter))
   986→    if (filtered.length === 0) return <FilterEmpty filter={filter} />
   987→    return <GenericTable items={filtered} depth={0} />
   988→  }
   989→  if (!matches(data, filter)) return <FilterEmpty filter={filter} />
   990→  return <GenericValue value={data} depth={0} />
   991→}
