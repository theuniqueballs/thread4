   159→function canonLineOf(oc: OCLocks): string {
   160→  const parts: string[] = [`${oc.name} —`]
   161→  if (oc.hair) parts.push(String(oc.hair).split('(')[0].trim())
   162→  if (oc.eyes) parts.push(String(oc.eyes).split('(')[0].trim())
   163→  if (oc.skin) parts.push(`${String(oc.skin).split('(')[0].trim()} skin`)
   164→  const marks = Array.isArray(oc.signature_marks) ? (oc.signature_marks as unknown[]).map(String) : []
   165→  for (const m of marks.slice(0, 4)) parts.push(m.split('(')[0].trim())
   166→  return parts.filter(Boolean).join(', ')
   167→}

   169→function canonTagHints(oc: OCLocks): string {
   170→  const marks = Array.isArray(oc.signature_marks) ? (oc.signature_marks as unknown[]).map(String) : []
   171→  const shield = Array.isArray(oc.anti_shield) ? (oc.anti_shield as unknown[]).map(String) : []
   172→  return [
   173→    `hair: ${String(oc.hair ?? '')}`,
   174→    `eyes: ${String(oc.eyes ?? '')}`,

   183→function paletteLine(p: Palette | undefined): string {
   184→  if (!p) return '—'
   185→  const dom = (p.dominant ?? []).slice(0, 3).join(', ')
   186→  const acc = [...(p.accent1 ?? []), ...(p.accent2 ?? []), ...(p.accent ?? [])].slice(0, 2).join(', ')
   187→  return `${p.name} (dominants: ${dom}${acc ? `; §51 accent: ${acc}` : ''}${p.light_type ? `; light: ${p.light_type}` : ''})`
   188→}

   190→function raceLine(race: Race | undefined, raceFeature: string | undefined): string {

   196→function slotFrame(slot: SlotPlan, ctx: ScribeCtx): string {
   197→  const lines: string[] = []
   198→  const recipe = ctx.recipes[recipeKeyOf(slot.rating)]
   199→  const carrier = ctx.carriers
   200→  lines.push(`SLOT P${String(slot.position).padStart(2, '0')} · genre ${slot.kind}${slot.exploratory ? ' · EXPLORATORY (probe a non-floor law, name it in THESIS)' : ''} · rating ${slot.rating}`)
   201→  if (slot.kind === 'OC' && slot.oc) {
   202→    const oc = ctx.ocs[slot.oc]
   203→    lines.push(`oc: ${slot.oc}${slot.ocTheme ? ` — author's theme for her: ${slot.ocTheme}` : ' — derive her theme from the batch law'}`)
   204→    lines.push(`NAME LAW: the name «${slot.oc}» NEVER appears in POS — not as a tag, not in prose (names trigger the renderer's foreign characters; author's order T4-04). Describe her by the descriptor tags below only.`)
   205→    if (oc) lines.push(`canon (locks exact, as DESCRIPTOR tags — no name):\n${canonTagHints(oc)}`)
   206→  } else {
   207→    lines.push(`who: ${raceLine(ctx.races.get(slot.race ?? ''), slot.raceFeature)}`)
   208→  }
   209→  lines.push(`pose: ${slot.pose} ${slot.poseName} (risk ${slot.poseRisk}) — the pose is ASSIGNED, honor it, no standing default`)
   210→  lines.push(`palette: ${paletteLine(ctx.palettes.get(slot.palette))}`)
   211→  const carriers = slot.carriers
   212→    .map((c) => {
   213→      const full = carrier.get(c.id)
   214→      return `  ${c.id} ${c.name} (deg ${c.deg})${full?.carrier ? ` — "${full.carrier}"` : ''}`
   215→    })
   216→    .join('\n')
   217→  lines.push(`carriers — weave ALL of these into the frame, one per mechanism group:\n${carriers}`)
   218→  const k = ctx.kNames.get(slot.kinetics[0] ?? '')
   219→  lines.push(`kinetic: ${slot.kinetics[0] ?? '—'}${k ? ` ${k.name}${k.light ? ` (${k.light})` : ''}` : ''}`)
   220→  lines.push(`lead zone (FIRST read of the silhouette): ${slot.lead}`)
   221→  lines.push(`register: ${slot.register} — voice in prose ONLY, never tags`)
   222→  lines.push(`closer: end the POS prose as a ${slot.closer} — ${closerHint(slot.closer)}`)
   223→  if (slot.witness) lines.push(`witness: the ${slot.witness} — one object that holds the frame's law`)
   224→  if (recipe) {
   225→    const hardSignals = recipe.signals_hard ?? []
   226→    lines.push(`REQUIRED rating signals — include ≥${recipe.signal_min} of these IN THE TAG BLOCK (before the environment tags): ${recipe.signals.slice(0, 14).join(', ')}`)
   227→    if (hardSignals.length > 0) {
   228→      lines.push(
   229→        `HARD CLAIM (RENDER LAW v1.2.0, render verdict T4-03): ≥${recipe.hard_min ?? 1} RENDER-PROVEN claim in the tag block — ${hardSignals.join(', ')}. cameltoe is NOT a claim (0 renders ever — flavor only). Through-fabric claims REQUIRE the fabric state tags (wet clothes / see-through) on ONE thin LIGHT garment, NO underlayer on the zone (bra/camisole/bandeau under sheer renders the underlayer and the tier dies to R). Dead fabrics (jeans/denim/leather/velvet/sweatpants/breeches/suit) never carry the claim.`
   230→      )
   231→      lines.push(
   232→        `CLAIM ZONE = the lead zone (${slot.lead}), OPEN TO CAMERA, single layer + the claim target. Delivery is stochastic (~1/8): build the best-odds frame — the claim and its fabric state EARLY in the tag run, the prose naming the same garment the tags claim. «Looks erotic but does nothing» is the failure this law kills.`
   233→      )
   234→    }
   235→    lines.push(`tier boundary: do NOT use higher-tier signals (${higherTierSignals(slot.rating, ctx.recipes).join(', ')}) — the counter-negatives are added by the machine`)
   236→  }
   237→  return lines.join('\n')
   238→}

   240→function closerHint(closer: string): string {
   241→  switch (closer) {
   242→    case 'dialogue': return 'a short spoken line at the end'
   243→    case 'long-fused': return 'a long-fuse image that keeps burning after the frame'
   244→    case 'fragment-pair': return 'two sentence fragments, paired'
   245→    case 'image-close': return 'a still image-read, the frame settling'
   246→    case 'action-close': return 'the motion caught mid-beat'
   247→    default: return 'a clean final image'
   248→  }
   249→}

   251→function higherTierSignals(tier: string, recipes: Record<string, TierRecipe>): string[] {
   252→  const order: Record<string, number> = { 'PG-13': 0, R: 1, 'R+': 2, X: 3 }
   253→  const mine = order[tier] ?? 0
   254→  const out: string[] = []
   255→  for (const [key, rec] of Object.entries(recipes)) {
   256→    const t = key === 'PG13' ? 'PG-13' : key === 'RPLUS' ? 'R+' : key
   257→    if ((order[t] ?? 0) > mine && t !== 'XXX') out.push(...(rec.signals ?? []).slice(0, 8))
   258→  }
   259→  return [...new Set(out)]
   260→}

   266→function assembleNeg(slot: SlotPlan, out: ScribeSlotOutput, ctx: ScribeCtx): string {
   267→  const f = ctx.floors
   268→  const terms: string[] = []
   269→  terms.push(...(f.face_guard ?? []))
   270→  terms.push(...(f.anti_loli ?? []))
   271→  terms.push(...(f.genital_lock ?? []), 'sex', 'sexual act')
   272→  terms.push(...(f.leak_guard ?? []))
   273→  const posLower = out.pos.toLowerCase()
   274→  if (!/candle|lantern|torch|chandelier|lamp|brazier/.test(posLower)) {
   275→    terms.push(...(f.candle_guard ?? []))
   276→  }
   277→  terms.push('male', 'man', '1boy', '2girls')
   278→  const recipe = ctx.recipes[recipeKeyOf(slot.rating)]
   279→  if (recipe) {
   280→    // МЕХАНИЗМ-ОСОЗНАННЫЙ КОНТР-NEG (вердикт T4-03: «только купальник»):
   281→    // активный on-skin механизм (taped nipples / handbra) поднимает
   282→    // «topless, naked breasts» из NEG — иначе рендерер закрывает грудь и
   283→    // душит сигнал. Сквозь-ткань механизмы держат полный грудной NEG.
   284→    let counter = [...(recipe.counter_neg ?? [])]
   285→    for (const m of Object.values(recipe.mechanisms ?? {})) {
   286→      const hard = m.hard ?? []
   287→      const active = hard.some((h) => posLower.includes(h.toLowerCase()))
   288→      if (active) {
   289→        counter = counter.filter(
   290→          (t) => !(m.counter_neg_lifted ?? []).some((x) => x.toLowerCase() === t.toLowerCase())
   291→        )
   292→      }
   293→    }
   294→    terms.push(...counter)
   295→  }
   296→  if (slot.kind === 'OC' && slot.oc) {
   297→    const oc = ctx.ocs[slot.oc]
   298→    if (oc && Array.isArray(oc.anti_shield)) {
   299→      terms.push(...(oc.anti_shield as unknown[]).map(String))
   300→    }
   301→  }
   302→  if (out.negExtra.trim()) {
   303→    for (const t of out.negExtra.split(/[,;]+/)) {
   304→      const x = t.trim()
   305→      if (x) terms.push(x)
   306→    }
   307→  }
   308→  // dedupe (case-insensitive), keep first spelling
   309→  const seen = new Set<string>()
   310→  const final: string[] = []
   311→  for (const t of terms) {
   312→    const k = t.toLowerCase()
   313→    if (seen.has(k)) continue
   314→    seen.add(k)
   315→    final.push(t)
   316→  }
   317→  return final.join(', ')
   318→}

   324→function normalizePos(pos: string, slot: SlotPlan, ctx: ScribeCtx): string {
   325→  let p = pos.replace(/\s*\n\s*/g, ' ').replace(/\s{2,}/g, ' ').trim()
   326→  // strip accidental markdown fences
   327→  p = p.replace(/^```[a-z]*\s*/i, '').replace(/```\s*$/, '').trim()
   328→  // ЗАПРЕТ ИМЁН (приказ автора, T4-04): имена ОС в POS — триггеры чужих
   329→  // персонажей у рендерера. Кодовая гварда: имя вычищается из POS всегда,
   330→  // даже если LLM его написал. Имена собственные, case-sensitive
   331→  // («ash-grey» не трогаем)
   332→  for (const name of Object.keys(ctx.ocs)) {
   333→    p = p
   334→      .replace(new RegExp(`\\b${name}\\b,\\s*`, 'g'), '') // «Miyu, tag» → «tag»
   335→      .replace(new RegExp(`,\\s*\\b${name}\\b`, 'g'), '') // «tag, Miyu» → «tag»
   336→      .replace(new RegExp(`\\s+\\b${name}\\b`, 'g'), '') // проза «… Miyu» → «…»
   337→  }
   338→  p = p
   339→    .replace(/\s{2,}/g, ' ')
   340→    .replace(/,\s*,/g, ',')
   341→    .replace(/,\s*\./g, '.')
   342→    .replace(/^\s*,\s*/, '')
   343→    .trim()
   344→  if (!/anime style/i.test(p)) {
   345→    p = `${recipeOpener(ctx.recipes, slot.rating)}, ${p}`
   346→  }
   347→  if (!/masterpiece/i.test(p)) {
   348→    p = `${p.replace(/[.\s]*$/, '')}. Masterpiece, best quality, anime artstyle.`
   349→  }
   350→  return p
   351→}

   366→  L.push(ctx.batchThesis.trim())
   367→  L.push('')
   368→  L.push('ЖАНРЫ (легенда — вердикт T4-02: жанр обязан быть виден): **OC** — канон-локи персонажа, тема в слоте; **NICHE** — невозможный образ: раса делает ФИЗИЧЕСКУЮ работу в кадре, свидетель держит кадр, невозможное — первое считывание силуэта; **VOLT** — плоть: камера-участник, тело в движении, взгляд-вектор, экспозиция тегом; **EXQUISITE** — ультра своего жанра.')
   369→  const mains = c.slots.filter((s) => s.kind !== 'OC')
   370→  const spread: Record<string, number> = {}
   371→  for (const s of mains) spread[s.rating] = (spread[s.rating] ?? 0) + 1
   372→  L.push(`СТРУКТУРА (закон 24): ${c.slots.length} промпта = 3 OC (P01-P03) + 21 мейн (P04-P24) · спред мейнов ${Object.entries(spread).map(([r, n]) => `${r}×${n}`).join(' · ')} · расовый каст ${c.racialCount}/21 · регистры третями.`)
   373→  L.push('RENDER PROTOCOL: PH ON — тег-блок канал правды, проза несёт красоту. Спайн назначен контрактом до первого слова; гейты читают те же числа.')
   374→  L.push('')
   375→  const acts = ctx.acts.length === 3 ? ctx.acts : ['ACT I', 'ACT II', 'ACT III']
   376→  const actRanges: [number, number][] = [[1, 8], [9, 16], [17, 24]]
   377→  for (let a = 0; a < 3; a++) {
   378→    L.push('════════════════════════════════════════════════════════════════════════')
   379→    L.push(`ACT ${['I', 'II', 'III'][a]} — ${acts[a].toUpperCase()}`)
   380→    L.push('════════════════════════════════════════════════════════════════════════')
   381→    L.push('')
   382→    for (let pos = actRanges[a][0]; pos <= actRanges[a][1]; pos++) {
   383→      const slot = c.slots[pos - 1]
   384→      const out = outputs.get(pos)
   385→      if (!slot || !out) continue
   386→      const who = slot.kind === 'OC' ? slot.oc : stripRaceParens(slot.race)
   387→      L.push(`P${String(pos).padStart(2, '0')} — ${out.anchor} (${slot.kind}${slot.exploratory ? ' ⚗' : ''} · ${who} · ${slot.rating} · ${slot.pose} · ${slot.palette})`)
   388→      L.push(`THESIS: ${out.thesis}`)
   389→      if (slot.kind === 'OC' && slot.oc && ctx.ocs[slot.oc]) {
   390→        L.push(`Canon: ${canonLineOf(ctx.ocs[slot.oc])}`)
   391→      }
   392→      const k = ctx.kNames.get(slot.kinetics[0] ?? '')
   393→      const spineParts = [
   394→        `${slot.pose} ${slot.poseName}`,
   395→        `${slot.kinetics[0] ?? '—'}${k ? ` ${k.name}` : ''}`,
   396→        `LEAD ${slot.lead}`,
   397→        `${slot.register} register`,
   398→        `closer ${slot.closer}`,
   399→      ]
   400→      if (slot.witness) spineParts.push(`witness: ${slot.witness}`)
   401→      if (slot.ocTheme && slot.kind === 'OC') spineParts.push(`OC theme: ${slot.ocTheme}`)
   402→      L.push(`Spine: ${spineParts.join(' · ')}`)
   403→      L.push(`Stack: ${slot.carriers.map((x) => x.id).join(' + ')}`)
   404→      L.push('POS:')
   405→      L.push('')
   406→      L.push(normalizePos(out.pos, slot, ctx))
   407→      L.push('')
   408→      L.push('NEG:')
   409→      L.push('')
   410→      L.push(assembleNeg(slot, out, ctx))
   411→      L.push('')
   412→    }
   413→  }
   414→  return L.join('\n').trimEnd() + '\n'
   415→}

   421→type ZAiChat = Awaited<ReturnType<typeof ZAI.create>>
   422→
   423→async function chat(zai: ZAiChat, system: string, user: string, log: string[], retries = 2): Promise<string> {
   424→  let lastErr: unknown = null
   425→  for (let attempt = 0; attempt <= retries; attempt++) {
   426→    try {
   427→      const completion = await zai.chat.completions.create({
   428→        messages: [
   429→          { role: 'assistant', content: system },
   430→          { role: 'user', content: user },
   431→        ],
   432→        thinking: { type: 'disabled' },
   433→      })
   434→      const content = completion.choices[0]?.message?.content ?? ''
   435→      if (content.trim().length > 0) return content
   436→      lastErr = new Error('empty completion')
   437→    } catch (e) {
   438→      lastErr = e
   439→    }
   440→    log.push(`  · LLM-вызов не удался (${attempt + 1}/${retries + 1})${attempt < retries ? ' — повтор' : ''}`)
   441→  }
   442→  throw lastErr instanceof Error ? lastErr : new Error('LLM failed')
   443→}

   445→function parseSlots(raw: string): Map<number, ScribeSlotOutput> {
   446→  const out = new Map<number, ScribeSlotOutput>()
   447→  const blocks = raw.split(/^###\s*P(\d{1,2})\s*$/m)
   448→  // split with capture: [pre, n1, body1, n2, body2, ...]
   449→  for (let i = 1; i + 1 < blocks.length + 1; i += 2) {
   450→    const n = parseInt(blocks[i], 10)
   451→    const body = blocks[i + 1] ?? ''
   452→    if (!Number.isFinite(n)) continue
   453→    const grab = (key: string): string => {
   454→      const re = new RegExp(`^${key}:\\s*(.*)$`, 'm')
   455→      const m = re.exec(body)
   456→      if (m) return m[1].trim()
   457→      // multiline POS: key: then lines until next KEY: line
   458→      const reM = new RegExp(`^${key}:\\s*\\n([\\s\\S]*?)(?=\\n[A-Z-]+:|$)`, 'm')
   459→      const mM = reM.exec(body)
   460→      return mM ? mM[1].trim() : ''
   461→    }
   462→    const posSingle = grab('POS')
   463→    const posMulti = (() => {
   464→      const re = /^POS:\s*\n?([\s\S]*?)(?=\nNEG-EXTRA:|$)/m.exec(body)
   465→      return re ? re[1].trim() : ''
   466→    })()
   467→    out.set(n, {
   468→      position: n,
   469→      anchor:
   470→        grab('ANCHOR').replace(/[^\w\s-]/g, '').trim().replace(/\s+/g, '-').toLowerCase().slice(0, 60) ||
   471→        grab('THESIS').split(/\s+/).slice(0, 3).join('-').toLowerCase().replace(/[^\w-]/g, '') ||
   472→        'unnamed-frame',
   473→      thesis: grab('THESIS'),
   474→      pos: (posSingle && posSingle.length > posMulti.length ? posSingle : posMulti) || posSingle || posMulti,
   475→      negExtra: grab('NEG-EXTRA'),
   476→    })
   477→  }
   478→  return out
   479→}

   485→export async function scribeBatch(
   486→  slug: string,
   487→  options: ScribeOptions = {}
   488→): Promise<ScribeResult> {
   489→  const log: string[] = []
   490→  const maxRepairRounds = options.maxRepairRounds ?? 2
   491→  const chunkSize = options.chunkSize ?? 6
   492→  const say = (s: string) => {
   493→    log.push(s)
   494→    options.onLog?.(s)
   495→  }
   496→
   497→  /* contract + specs */
   498→  const contract = readJson<BatchContract>(path.join(CONTRACTS_DIR, `${slug}.json`))
   499→  if (!contract) throw new Error(`контракт ${slug} не найден — сначала сборка`)
   500→  // guard: сданный батч не перезаписывается черновиком (first-run-clean свято)
   501→  const allDelivered = readEvents()
   502→    .filter((e) => e.type === 'batch.delivered')
   503→    .map((e) => String(e.data?.slug ?? ''))
   504→  if (allDelivered.includes(slug)) {
   505→    throw new Error(`${slug} уже сдан — черновик не перезаписывает сданный батч`)
   506→  }
   507→  const carriersSpec = getCarriers()
   508→  const posesSpec = getPoses()
   509→  const palettesSpec = getPalettes()
   510→  const racesSpec = getRaces()
   511→  const ocCanon = getOCCanon()
   512→  const recipesSpec = getRatingRecipes()
   513→  const bans = getBans()
   514→  if (!carriersSpec || !posesSpec || !palettesSpec || !racesSpec || !ocCanon || !recipesSpec || !bans) {
   515→    throw new Error('спеки не загружены')
   516→  }
   517→  const poolsSpec = readJson<{ sections: Record<string, { id: string; name?: string; light?: string }[]> }>(
   518→    path.join(process.cwd(), 'thread4', 'specs', 'pools.json')
   519→  )
   520→  const kNames = new Map<string, { name: string; light: string }>()
   521→  for (const k of poolsSpec?.sections?.kinetics_k ?? []) {
   522→    kNames.set(k.id, { name: k.name ?? k.id, light: k.light ?? '' })
   523→  }
   524→
   525→  const ctx: ScribeCtx = {
   526→    contract,
   527→    carriers: new Map(Object.values(carriersSpec.classes).flat().map((c) => [c.id, c])),
   528→    poses: new Map(posesSpec.poses.map((p) => [p.id, p])),
   529→    palettes: new Map(palettesSpec.palettes.map((p) => [p.id, p])),
   530→    races: new Map(racesSpec.races.map((r) => [r.name, r])),
   531→    ocs: ocCanon.ocs,
   532→    recipes: recipesSpec.tiers as unknown as Record<string, TierRecipe>,
   533→    floors: recipesSpec.eternal_floors as unknown as Record<string, string[]>,
   534→    kNames,
   535→    batchThesis: '',
   536→    acts: [],
   537→  }
   538→
   539→  const engine = contract.engineLaw
   540→  const theme = contract.theme
   541→
   542→  say(`Писец: ${slug} «${theme}» · движок ${contract.engine} · ${contract.slots.length} слотов`)
   543→
   544→  const zai = await ZAI.create()
   545→
   546→  /* ---- call 0: batch spine (title, thesis, acts) ---- */
   547→  say('Шаг 1/4: закон батча, название, три акта…')
   548→  const spineRaw = await chat(
   549→    zai,
   550→    SYSTEM_PROMPT,
   551→    `BATCH ${slug} «${theme}». ENGINE LAW: ${engine}\n\nPlan the batch spine. The theme rides the engine's axis and stays there. Return EXACTLY, nothing else:\nTITLE: <2-5 words, no quotes inside>\nTHESIS: <one paragraph, 90-140 words: the batch's ONE law, physical and testable in-frame, how it is delivered across the 24 frames, and how the three acts escalate it — no narrated morality, the law lives in fabric and silhouette>\nACT I: <act name, 2-5 words>\nACT II: <act name>\nACT III: <act name>`,
   552→    log
   553→  )
   554→  const title = (/^TITLE:\s*(.+)$/m.exec(spineRaw)?.[1] ?? theme).replace(/^["«]|["»]$/g, '').trim().slice(0, 80) || theme
   555→  const thesisP = (/^THESIS:\s*\n?([\s\S]*?)(?=\nACT I:|$)/m.exec(spineRaw)?.[1] ?? '').trim()
   556→  ctx.batchThesis = thesisP || `The batch's law: ${engine}`
   557→  ctx.acts = [
   558→    (/^ACT I:\s*(.+)$/m.exec(spineRaw)?.[1] ?? 'The Law Arrives').trim(),
   559→    (/^ACT II:\s*(.+)$/m.exec(spineRaw)?.[1] ?? 'The Law Worn In').trim(),
   560→    (/^ACT III:\s*(.+)$/m.exec(spineRaw)?.[1] ?? 'What the Law Keeps').trim(),
   561→  ]
   562→  say(`  · «${title}» · акты: ${ctx.acts.join(' / ')}`)
   563→
   564→  /* ---- calls 1..N: slot chunks ---- */
   565→  const outputs = new Map<number, ScribeSlotOutput>()
   566→  const chunks: SlotPlan[][] = []
   567→  for (let i = 0; i < contract.slots.length; i += chunkSize) {
   568→    chunks.push(contract.slots.slice(i, i + chunkSize))
   569→  }
   570→  const batchLawBlock = `BATCH ${slug} «${title}» — theme: ${theme}\nENGINE LAW: ${engine}\nBATCH THESIS (yours, keep it): ${ctx.batchThesis}\nGENRE PLAN: ${contract.slots.filter((s) => s.kind === 'NICHE').length} NICHE (R) · ${contract.slots.filter((s) => s.kind === 'VOLT' || s.kind === 'EXQUISITE').length} VOLT/EXQUISITE (R+ incl. 3 OC) · 2 X`
   571→
   572→  for (let ci = 0; ci < chunks.length; ci++) {
   573→    const chunk = chunks[ci]
   574→    const positions = chunk.map((s) => `P${String(s.position).padStart(2, '0')}`).join(', ')
   575→    say(`Шаг 2/4: пишу слоты ${positions} (чанк ${ci + 1}/${chunks.length})…`)
   576→    const user = `${batchLawBlock}\n\nSLOTS (write ALL of them, in order):\n\n${chunk.map((s) => slotFrame(s, ctx)).join('\n\n')}`
   577→    const raw = await chat(zai, SYSTEM_PROMPT, user, log)
   578→    const parsed = parseSlots(raw)
   579→    let missing = chunk.filter((s) => !parsed.has(s.position) || !(parsed.get(s.position)?.pos ?? '').trim())
   580→    if (missing.length > 0) {
   581→      say(`  · чанк ${ci + 1}: пропущены ${missing.map((s) => s.position).join(', ')} — повтор`)
   582→      const retry = await chat(
   583→        zai,
   584→        SYSTEM_PROMPT,
   585→        `${user}\n\nREMINDER: output EVERY slot above in the exact format. Do not skip any.`,
   586→        log
   587→      )
   588→      for (const [n, o] of parseSlots(retry)) parsed.set(n, o)
   589→      missing = chunk.filter((s) => !parsed.has(s.position) || !(parsed.get(s.position)?.pos ?? '').trim())
   590→    }
   591→    for (const s of chunk) {
   592→      const o = parsed.get(s.position)
   593→      if (o && o.pos.trim()) outputs.set(s.position, o)
   594→    }
   595→    if (missing.length > 0) {
   596→      say(`  · чанк ${ci + 1}: НЕ написаны ${missing.map((s) => s.position).join(', ')} — уйдут в ремонт`)
   597→    }
   598→  }
   599→
   600→  /* ---- assembly + gates + repair ---- */
   601→  let markdown = assembleBatch(ctx, outputs, title)
   602→  let result: GatesResult | null = null
   603→  let rounds = 0
   604→  const batchPath = path.join(BATCHES_DIR, `${slug}.md`)
   605→
   606→  for (let round = 0; round <= maxRepairRounds; round++) {
   607→    writeText(batchPath, markdown)
   608→    say(`Шаг 3/4: гейты (сухой прогон, круг ${round + 1})…`)
   609→    result = runGates(slug, true)
   610→    if (!result) throw new Error('гейты не нашли файл батча')
   611→    const hardFails = result.receipts.filter((r) => r.level === 'hard' && r.verdict === 'FAIL')
   612→    const missingSlots = contract.slots.filter((s) => !outputs.has(s.position))
   613→    if (hardFails.length === 0 && missingSlots.length === 0) {
   614→      say(`  · круг ${round + 1}: hard PASS${round === 0 ? ' — чистый первый черновик' : ` — после ${round} круг(ов) ремонта`}`)
   615→      break
   616→    }
   617→    if (round === maxRepairRounds) {
   618→      say(`  · круг ${round + 1}: hard FAIL остался — сдача невозможна, черновик записан`)
   619→      break
   620→    }
   621→    rounds = round + 1
   622→    say(`  · круг ${round + 1}: hard FAIL — ремонт:`)
   623→    for (const r of hardFails) {
   624→      for (const fnd of r.findings.slice(0, 6)) say(`    [${r.gate}] ${fnd}`)
   625→    }
   626→    // failed positions from findings + missing slots
   627→    const failed = new Set<number>(missingSlots.map((s) => s.position))
   628→    for (const r of hardFails) {
   629→      for (const fnd of r.findings) {
   630→        const m = /P(\d{1,2})/.exec(fnd)
   631→        if (m) {
   632→          const p = parseInt(m[1], 10)
   633→          if (p >= 1 && p <= contract.slots.length) failed.add(p)
   634→        }
   635→      }
   636→    }
   637→    if (failed.size === 0) break // nothing slot-specific to repair
   638→    const repairSlots = contract.slots.filter((s) => failed.has(s.position))
   639→    say(`  · ремонт слотов: ${[...failed].sort((a, b) => a - b).join(', ')}`)
   640→    const repairUser = `${batchLawBlock}\n\nREWRITE THESE SLOTS — they failed the machine gates. Keep what worked, fix what is flagged. Same exact format.\n\n${repairSlots.map((s) => slotFrame(s, ctx)).join('\n\n')}\n\nGATE FAILURES TO FIX:\n${hardFails.flatMap((r) => r.findings.map((f) => `- [${r.gate}] ${f}`)).slice(0, 24).join('\n')}`
   641→    const raw = await chat(zai, SYSTEM_PROMPT, repairUser, log)
   642→    for (const [n, o] of parseSlots(raw)) {
   643→      if (o.pos.trim()) outputs.set(n, o)
   644→    }
   645→    markdown = assembleBatch(ctx, outputs, title)
   646→  }
   647→
   648→  if (!result) throw new Error('гейты не прогнались')
   649→
   650→  /* ---- record + event ---- */
   651→  say(`Шаг 4/4: черновик записан → batches/${slug}.md`)
   652→  appendEvent(
   653→    'scribe.drafted',
   654→    `${slug} «${title}»: авто-писец написал черновик — ${outputs.size}/${contract.slots.length} слотов, ремонт ${rounds} круг(а), гейты dry ${result.hardPass ? 'PASS' : 'FAIL'} (sha ${result.sha10})`,
   655→    {
   656→      slug,
   657→      title,
   658→      slots: outputs.size,
   659→      rounds,
   660→      hardPass: result.hardPass,
   661→      sha10: result.sha10,
   662→      engine: contract.engine,
   663→    }
   664→  )
   665→
   666→  return {
   667→    slug,
   668→    title,
   669→    rounds,
   670→    hardPass: result.hardPass,
   671→    firstRunClean: result.firstRunClean,
   672→    sha10: result.sha10,
   673→    receipts: result.receipts,
   674→    failedSlots: contract.slots.filter((s) => !outputs.has(s.position)).map((s) => s.position),
   675→    log,
   676→    markdown,
   677→  }
   678→}
