   261→function CompileTab() {
   262→  const [theme, setTheme] = useState('')
   263→  const [engine, setEngine] = useState('')
   264→  const [ocRows, setOcRows] = useState([
   265→    { name: '', theme: '' },
   266→    { name: '', theme: '' },
   267→    { name: '', theme: '' },
   268→  ])
   269→  const [busy, setBusy] = useState(false)
   270→  const [result, setResult] = useState<CompilePayload | null>(null)
   271→  const [err, setErr] = useState<string | null>(null)
   272→  const [copied, setCopied] = useState<string | null>(null)
   273→  const contracts = useApi<{ items: { slug: string; theme: string; createdAt: string }[] }>('/api/t4/contracts')
   274→  const [viewSlug, setViewSlug] = useState<string | null>(null)
   275→  const detail = useApi<{ markdown: string }>(viewSlug ? `/api/t4/contracts/${viewSlug}` : null)
   276→
   277→  async function compile() {
   278→    const t = theme.trim()
   279→    if (!t || busy) return
   280→    setBusy(true)
   281→    setErr(null)
   282→    setResult(null)
   283→    const ocOrders = ocRows.map((r) => r.name.trim()).filter(Boolean)
   284→    const ocThemes: Record<string, string> = {}
   285→    for (const r of ocRows) {
   286→      const n = r.name.trim()
   287→      const th = r.theme.trim()
   288→      if (n && th) ocThemes[n] = th
   289→    }
   290→    const payload: Record<string, unknown> = { theme: t }
   291→    if (ocOrders.length > 0) payload.ocOrders = ocOrders
   292→    if (Object.keys(ocThemes).length > 0) payload.ocThemes = ocThemes
   293→    if (engine.trim()) payload.engine = engine.trim()
   294→    try {
   295→      const res = await postJson<CompilePayload>('/api/t4/compile', payload)
   296→      setResult(res)
   297→      contracts.reload()
   298→    } catch (e) {
   299→      setErr(e instanceof ApiError ? `API: ${e.message}` : 'Сеть недоступна')
   300→    } finally {
   301→      setBusy(false)
   302→    }
   303→  }
   304→
   305→  async function copyOrder(slug: string) {
   306→    try {
   307→      await navigator.clipboard.writeText(`Super Z, произведи ${slug}`)
   308→      setCopied(slug)
   309→      setTimeout(() => setCopied((c) => (c === slug ? null : c)), 2500)
   310→    } catch {
   311→      setCopied(null)
   312→    }
   313→  }
   314→
   315→  const items = contracts.data?.items ?? []
   316→
   317→  return (
   318→    <div className="space-y-6">
   319→      <Panel title="Компилятор батча" icon={<FlaskConical className="size-4" />}>
   320→        <div className="space-y-3">
   321→          <p className="text-xs leading-relaxed text-zinc-500">
   322→            Тема от автора → слот-план 21+3 OC: жанры, рейтинги по рецептуре, назначенные носители, позы,
   323→            палитры, расы, регистры. Диверсия назначается ДО письма (не ловится линтами после).
   324→            Контракт = экспозиция для автора + закон для писца.
   325→          </p>
   326→          <Textarea
   327→            value={theme}
   328→            onChange={(e) => setTheme(e.target.value)}
   329→            placeholder="Тема батча — например: «город, где усталость носит как меха»…"
   330→            className="min-h-20 border-zinc-800 bg-zinc-950 text-sm"
   331→          />
   332→          <div className="grid gap-2 sm:grid-cols-[180px_1fr]">
   333→            <Input
   334→              value={engine}
   335→              onChange={(e) => setEngine(e.target.value)}
   336→              placeholder="движок (пусто = ротация)"
   337→              className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   338→            />
   339→            <p className="self-center text-[11px] leading-tight text-zinc-600">
   340→              tint · bespoke · counterfall · … — ключ из спеки движков; пусто = ротация.
   341→            </p>
   342→          </div>
   343→          <div className="space-y-2">
   344→            <p className="text-[11px] uppercase tracking-wider text-zinc-600">
   345→              Заказ OC — до трёх · имя из канона (пусто = ротация) · тема по желанию
   346→            </p>
   347→            {ocRows.map((row, i) => (
   348→              <div key={i} className="grid gap-2 sm:grid-cols-[180px_1fr]">
   349→                <Input
   350→                  value={row.name}
   351→                  onChange={(e) =>
   352→                    setOcRows((rows) => rows.map((r, j) => (j === i ? { ...r, name: e.target.value } : r)))
   353→                  }
   354→                  placeholder={i === 0 ? 'например: Lyn' : 'имя OC'}
   355→                  className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   356→                />
   357→                <Input
   358→                  value={row.theme}
   359→                  onChange={(e) =>
   360→                    setOcRows((rows) => rows.map((r, j) => (j === i ? { ...r, theme: e.target.value } : r)))
   361→                  }
   362→                  placeholder="тема/сценарий для неё — пусто = писец выведет из темы батча"
   363→                  className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   364→                />
   365→              </div>
   366→            ))}
   367→          </div>
   368→          <div className="flex items-center gap-3">
   369→            <Button
   370→              onClick={compile}
   371→              disabled={busy || theme.trim() === ''}
   372→              className="bg-amber-500 text-zinc-950 hover:bg-amber-400 disabled:opacity-40"
   373→            >
   374→              {busy ? 'Компилирую…' : 'Скомпилировать'}
   375→            </Button>
   376→            {err ? <span className="text-xs text-rose-400">{err}</span> : null}
   377→          </div>
   378→        </div>
   379→      </Panel>
   380→
   381→      {result ? (
   382→        <Panel title={`Контракт ${result.slug}`} icon={<ScrollText className="size-4" />} bodyClassName="max-h-[70vh] overflow-y-auto t4-scroll">
   383→          <MarkdownView>{result.markdown ?? ''}</MarkdownView>
   384→        </Panel>
   385→      ) : null}
   386→
   387→      <Panel title="Производство — от сборки к батчу" icon={<Sparkles className="size-4" />}>
   388→        <ol className="space-y-2 text-xs leading-relaxed text-zinc-400">
   389→          <li>
   390→            <span className="text-amber-400">1 ·</span> Собери контракт выше — тема, при желании движок и заказ
   391→            OC с их темами. Сборка занимает секунды и ни к чему не обязывает: это план, не батч.
   392→          </li>
   393→          <li>
   394→            <span className="text-amber-400">2 ·</span> Нажми «Приказ» у нужного контракта в списке ниже —
   395→            строка <Mono>{'«Super Z, произведи T4-NN»'}</Mono> скопируется одной кнопкой. Темы ОС, если
   396→            свои, допиши прямо в приказ.
   397→          </li>
   398→          <li>
   399→            <span className="text-amber-400">3 ·</span> Отправь приказ в чат. Писец пишет 21 промпт + 3 OC
   400→            против контракта в одном файле, самопроверка по гейтам, официальный прогон, сдача с ворклогом —
   401→            батч и квитанции гейтов появятся во вкладке «Батчи».
   402→          </li>
   403→        </ol>
   404→      </Panel>
   405→
   406→      <Panel title="Контракты" icon={<FileText className="size-4" />}>
   407→        {contracts.loading ? (
   408→          <SkeletonBlock lines={3} />
   409→        ) : items.length === 0 ? (
   410→          <EmptyState title="Контрактов пока нет" hint="Первый контракт станет T4-01." />
   411→        ) : (
   412→          <div className="space-y-1.5">
   413→            {items.map((c) => (
   414→              <div key={c.slug} className="flex items-center justify-between gap-3 rounded-md border border-zinc-800 bg-zinc-900/60 px-3 py-2">
   415→                <button
   416→                  onClick={() => setViewSlug(viewSlug === c.slug ? null : c.slug)}
   417→                  className="flex min-w-0 items-center gap-2 text-left"
   418→                >
   419→                  <Mono>{c.slug}</Mono>
   420→                  <span className="truncate text-xs text-zinc-300">{c.theme}</span>
   421→                </button>
   422→                <div className="flex shrink-0 items-center gap-2">
   423→                  <button
   424→                    onClick={() => copyOrder(c.slug)}
   425→                    title="Скопировать приказ на производство"
   426→                    className="flex items-center gap-1 rounded-md border border-amber-500/40 bg-amber-500/10 px-2 py-1 text-[11px] text-amber-300 transition-colors hover:bg-amber-500/20"
   427→                  >
   428→                    {copied === c.slug ? <Check className="size-3" /> : <Copy className="size-3" />}
   429→                    {copied === c.slug ? 'Скопировано' : 'Приказ'}
   430→                  </button>
   431→                  <span className="text-[11px] text-zinc-600">{formatDate(c.createdAt)}</span>
   432→                </div>
   433→              </div>
   434→            ))}
   435→            {viewSlug ? (
   436→              <div className="mt-3 rounded-md border border-zinc-800 bg-zinc-950/60 p-4">
   437→                {detail.loading ? (
   438→                  <SkeletonBlock lines={8} />
   439→                ) : detail.error ? (
   440→                  <ErrorNote text="Контракт недоступен" />
   441→                ) : (
   442→                  <div className="max-h-[50vh] overflow-y-auto t4-scroll">
   443→                    <MarkdownView>{detail.data?.markdown ?? ''}</MarkdownView>
   444→                  </div>
   445→                )}
   446→              </div>
   447→            ) : null}
   448→          </div>
   449→        )}
   450→      </Panel>
   451→    </div>
   452→  )
   453→}

   577→function VerdictsTab() {
   578→  const [slug, setSlug] = useState('')
   579→  const [text, setText] = useState('')
   580→  const [busy, setBusy] = useState(false)
   581→  const [note, setNote] = useState<string | null>(null)
   582→
   583→  async function submit() {
   584→    if (busy || text.trim() === '') return
   585→    setBusy(true)
   586→    setNote(null)
   587→    try {
   588→      await postJson('/api/t4/events', {
   589→        type: 'render.verdict',
   590→        summary: `${slug.trim() || 'batch'}: ${text.trim().slice(0, 120)}`,
   591→        data: { slug: slug.trim(), prose: text.trim() },
   592→      })
   593→      setNote('Вердикт записан в лог событий.')
   594→      setText('')
   595→    } catch (e) {
   596→      setNote(e instanceof ApiError && isNotFound(e) ? 'API появится после интеграции ядра' : 'Не удалось записать')
   597→    } finally {
   598→      setBusy(false)
   599→    }
   600→  }
   601→
   602→  return (
   603→    <div className="space-y-6">
   604→      <Panel title="Вердикт автора" icon={<Heart className="size-4" />}>
   605→        <div className="space-y-3">
   606→          <p className="text-xs leading-relaxed text-zinc-500">
   607→            Авторская проза — самый сильный сигнал системы. Пиши как видишь; структура извлекается при
   608→            кодировании в taste-лог.
   609→          </p>
   610→          <Input
   611→            value={slug}
   612→            onChange={(e) => setSlug(e.target.value)}
   613→            placeholder="Слаг батча — T4-01"
   614→            className="border-zinc-800 bg-zinc-950 text-sm"
   615→          />
   616→          <Textarea
   617→            value={text}
   618→            onChange={(e) => setText(e.target.value)}
   619→            placeholder="«Охуенная поза решила всё…», «скатились в волт…» — как есть"
   620→            className="min-h-32 border-zinc-800 bg-zinc-950 text-sm"
   621→          />
   622→          <div className="flex items-center gap-3">
   623→            <Button
   624→              onClick={submit}
   625→              disabled={busy || text.trim() === ''}
   626→              className="bg-amber-500 text-zinc-950 hover:bg-amber-400 disabled:opacity-40"
   627→            >
   628→              {busy ? 'Записываю…' : 'Записать вердикт'}
   629→            </Button>
   630→            {note ? <span className="text-xs text-zinc-400">{note}</span> : null}
   631→          </div>
   632→        </div>
   633→      </Panel>
   634→
   635→      <Panel title="VLM-разбор рендера" icon={<Sparkles className="size-4" />}>
   636→        <EmptyState
   637→          title="VLM-петля — скоро"
   638→          hint="Загрузка рендера → автоматический разбор (силуэт / первый считыв / носители / scroll-stop). Автор разрешил попробовать — ядро подключит."
   639→        />
   640→      </Panel>
   641→    </div>
   642→  )
   643→}
