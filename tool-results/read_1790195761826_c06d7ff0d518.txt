   266→function CompileTab() {
   267→  const [theme, setTheme] = useState('')
   268→  const [engine, setEngine] = useState('')
   269→  const [ocRows, setOcRows] = useState([
   270→    { name: '', theme: '' },
   271→    { name: '', theme: '' },
   272→    { name: '', theme: '' },
   273→  ])
   274→  const [busy, setBusy] = useState(false)
   275→  const [result, setResult] = useState<CompilePayload | null>(null)
   276→  const [err, setErr] = useState<string | null>(null)
   277→  const [copied, setCopied] = useState<string | null>(null)
   278→  const contracts = useApi<{ items: { slug: string; theme: string; createdAt: string }[] }>('/api/t4/contracts')
   279→  const [viewSlug, setViewSlug] = useState<string | null>(null)
   280→  const detail = useApi<{ markdown: string }>(viewSlug ? `/api/t4/contracts/${viewSlug}` : null)
   281→
   282→  /* авто-писец */
   283→  const [scribe, setScribe] = useState<{
   284→    slug: string
   285→    busy: boolean
   286→    err: string | null
   287→    res: ScribeResponse | null
   288→  } | null>(null)
   289→  const [deliverNote, setDeliverNote] = useState<string | null>(null)
   290→
   291→  async function compile() {
   292→    const t = theme.trim()
   293→    if (!t || busy) return
   294→    setBusy(true)
   295→    setErr(null)
   296→    setResult(null)
   297→    const ocOrders = ocRows.map((r) => r.name.trim()).filter(Boolean)
   298→    const ocThemes: Record<string, string> = {}
   299→    for (const r of ocRows) {
   300→      const n = r.name.trim()
   301→      const th = r.theme.trim()
   302→      if (n && th) ocThemes[n] = th
   303→    }
   304→    const payload: Record<string, unknown> = { theme: t }
   305→    if (ocOrders.length > 0) payload.ocOrders = ocOrders
   306→    if (Object.keys(ocThemes).length > 0) payload.ocThemes = ocThemes
   307→    if (engine.trim()) payload.engine = engine.trim()
   308→    try {
   309→      const res = await postJson<CompilePayload>('/api/t4/compile', payload)
   310→      setResult(res)
   311→      contracts.reload()
   312→    } catch (e) {
   313→      setErr(e instanceof ApiError ? `API: ${e.message}` : 'Сеть недоступна')
   314→    } finally {
   315→      setBusy(false)
   316→    }
   317→  }
   318→
   319→  async function copyOrder(slug: string) {
   320→    try {
   321→      await navigator.clipboard.writeText(`Super Z, произведи ${slug}`)
   322→      setCopied(slug)
   323→      setTimeout(() => setCopied((c) => (c === slug ? null : c)), 2500)
   324→    } catch {
   325→      setCopied(null)
   326→    }
   327→  }
   328→
   329→  async function runScribe(slug: string) {
   330→    if (scribe?.busy) return
   331→    setDeliverNote(null)
   332→    setScribe({ slug, busy: true, err: null, res: null })
   333→    try {
   334→      const res = await postJson<ScribeResponse>('/api/t4/scribe', { slug })
   335→      setScribe({ slug, busy: false, err: null, res })
   336→    } catch (e) {
   337→      setScribe({
   338→        slug,
   339→        busy: false,
   340→        err: e instanceof ApiError ? `API: ${e.message}` : 'Сеть недоступна — писец не смог',
   341→        res: null,
   342→      })
   343→    }
   344→  }
   345→
   346→  async function deliverOfficial(slug: string) {
   347→    setDeliverNote(null)
   348→    try {
   349→      const res = await postJson<DeliverResponse>('/api/t4/gates', { slug, deliver: true })
   350→      setDeliverNote(
   351→        res.delivered
   352→          ? `${slug} «${res.title}» сдан официально${res.result?.firstRunClean ? ' · FIRST RUN CLEAN' : ''} — батч, мета и ворклог во вкладке «Батчи»`
   353→          : `${slug}: hard FAIL — черновик чини (см. квитанции во вкладке «Батчи»)`
   354→      )
   355→    } catch (e) {
   356→      setDeliverNote(e instanceof ApiError ? `API: ${e.message}` : 'Сеть недоступна')
   357→    }
   358→  }
   359→
   360→  const items = contracts.data?.items ?? []
   361→
   362→  return (
   363→    <div className="space-y-6">
   364→      <Panel title="Компилятор батча" icon={<FlaskConical className="size-4" />}>
   365→        <div className="space-y-3">
   366→          <p className="text-xs leading-relaxed text-zinc-500">
   367→            Тема от автора → слот-план 24 промпта (21 мейн + 3 OC — вердикт T4-02): жанры,
   368→            рейтинги по рецептуре, назначенные носители, позы, палитры, расы, регистры.
   369→            Диверсия назначается ДО письма. Контракт = экспозиция для автора + закон для писца.
   370→          </p>
   371→          <Textarea
   372→            value={theme}
   373→            onChange={(e) => setTheme(e.target.value)}
   374→            placeholder="Тема батча — например: «город, где усталость носит как меха»…"
   375→            className="min-h-20 border-zinc-800 bg-zinc-950 text-sm"
   376→          />
   377→          <div className="grid gap-2 sm:grid-cols-[180px_1fr]">
   378→            <Input
   379→              value={engine}
   380→              onChange={(e) => setEngine(e.target.value)}
   381→              placeholder="движок (пусто = ротация)"
   382→              className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   383→            />
   384→            <p className="self-center text-[11px] leading-tight text-zinc-600">
   385→              tint · bespoke · counterfall · … — ключ из спеки движков; пусто = ротация.
   386→            </p>
   387→          </div>
   388→          <div className="space-y-2">
   389→            <p className="text-[11px] uppercase tracking-wider text-zinc-600">
   390→              Заказ OC — до трёх · имя из канона (пусто = ротация) · тема по желанию
   391→            </p>
   392→            {ocRows.map((row, i) => (
   393→              <div key={i} className="grid gap-2 sm:grid-cols-[180px_1fr]">
   394→                <Input
   395→                  value={row.name}
   396→                  onChange={(e) =>
   397→                    setOcRows((rows) => rows.map((r, j) => (j === i ? { ...r, name: e.target.value } : r)))
   398→                  }
   399→                  placeholder={i === 0 ? 'например: Lyn' : 'имя OC'}
   400→                  className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   401→                />
   402→                <Input
   403→                  value={row.theme}
   404→                  onChange={(e) =>
   405→                    setOcRows((rows) => rows.map((r, j) => (j === i ? { ...r, theme: e.target.value } : r)))
   406→                  }
   407→                  placeholder="тема/сценарий для неё — пусто = писец выведет из темы батча"
   408→                  className="h-9 border-zinc-800 bg-zinc-950 text-xs"
   409→                />
   410→              </div>
   411→            ))}
   412→          </div>
   413→          <div className="flex items-center gap-3">
   414→            <Button
   415→              onClick={compile}
   416→              disabled={busy || theme.trim() === ''}
   417→              className="bg-amber-500 text-zinc-950 hover:bg-amber-400 disabled:opacity-40"
   418→            >
   419→              {busy ? 'Компилирую…' : 'Скомпилировать'}
   420→            </Button>
   421→            {err ? <span className="text-xs text-rose-400">{err}</span> : null}
   422→          </div>
   423→        </div>
   424→      </Panel>
   425→
   426→      {result ? (
   427→        <Panel title={`Контракт ${result.slug}`} icon={<ScrollText className="size-4" />} bodyClassName="max-h-[70vh] overflow-y-auto t4-scroll">
   428→          <MarkdownView>{result.markdown ?? ''}</MarkdownView>
   429→        </Panel>
   430→      ) : null}
   431→
   432→      <Panel title="Производство — от сборки к батчу" icon={<Sparkles className="size-4" />}>
   433→        <ol className="space-y-2 text-xs leading-relaxed text-zinc-400">
   434→          <li>
   435→            <span className="text-amber-400">1 ·</span> Собери контракт выше — тема, при желании движок и заказ
   436→            OC с их темами. Сборка занимает секунды и ни к чему не обязывает: это план, не батч.
   437→          </li>
   438→          <li>
   439→            <span className="text-amber-400">2а ·</span> <span className="text-zinc-200">Приказ в чате:</span> кнопка
   440→            «Приказ» у контракта копирует <Mono>{'«Super Z, произведи T4-NN»'}</Mono> — писец Super Z пишет
   441→            21 мейн + 3 OC, самопроверка, сдача с ворклогом.
   442→          </li>
   443→          <li>
   444→            <span className="text-amber-400">2б ·</span> <span className="text-zinc-200">Авто-писец:</span> кнопка
   445→            «Писец» у контракта — машина пишет черновик по тому же контракту (2-5 минут), гейты
   446→            гоняют его автоматически, ремонт до 2 кругов. Потом — «Сдать» или моя полировка.
   447→          </li>
   448→          <li>
   449→            <span className="text-amber-400">3 ·</span> Батч и квитанции гейтов — во вкладке «Батчи»; твой
   450→            вердикт — там же, одной панелью чипов.
   451→          </li>
   452→        </ol>
   453→      </Panel>
   454→
   455→      <Panel title="Контракты" icon={<FileText className="size-4" />}>
   456→        {contracts.loading ? (
   457→          <SkeletonBlock lines={3} />
   458→        ) : items.length === 0 ? (
   459→          <EmptyState title="Контрактов пока нет" hint="Первый контракт станет T4-01." />
   460→        ) : (
   461→          <div className="space-y-1.5">
   462→            {items.map((c) => (
   463→              <div key={c.slug} className="flex items-center justify-between gap-3 rounded-md border border-zinc-800 bg-zinc-900/60 px-3 py-2">
   464→                <button
   465→                  onClick={() => setViewSlug(viewSlug === c.slug ? null : c.slug)}
   466→                  className="flex min-w-0 items-center gap-2 text-left"
   467→                >
   468→                  <Mono>{c.slug}</Mono>
   469→                  <span className="truncate text-xs text-zinc-300">{c.theme}</span>
   470→                </button>
   471→                <div className="flex shrink-0 items-center gap-2">
   472→                  <button

   623→function QuickVerdict({ slug }: { slug: string }) {
   624→  const [verdict, setVerdict] = useState<string | null>(null)
   625→  const [issues, setIssues] = useState<string[]>([])
   626→  const [text, setText] = useState('')
   627→  const [busy, setBusy] = useState(false)
   628→  const [note, setNote] = useState<string | null>(null)
   629→
   630→  function toggleIssue(x: string) {
   631→    setIssues((cur) => (cur.includes(x) ? cur.filter((i) => i !== x) : [...cur, x]))
   632→  }
   633→
   634→  async function submit() {
   635→    if (busy || !verdict) return
   636→    setBusy(true)
   637→    setNote(null)
   638→    const vLabel = QUICK_VERDICTS.find((v) => v.id === verdict)?.label ?? verdict
   639→    const prose = [text.trim(), issues.length > 0 ? `Проблемы: ${issues.join(', ')}.` : '']
   640→      .filter(Boolean)
   641→      .join('\n\n')
   642→    try {
   643→      await postJson('/api/t4/events', {
   644→        type: 'render.verdict',
   645→        summary: `${slug}: ${vLabel}${issues.length > 0 ? ` · ${issues.join(', ')}` : ''}${text.trim() ? ` — ${text.trim().slice(0, 90)}` : ''}`,
   646→        data: { slug, verdict, tags: issues, prose },
   647→      })
   648→      setNote('Вердикт записан в лог — он кормит правки закона и вкус.')
   649→      setVerdict(null)
   650→      setIssues([])
   651→      setText('')
   652→    } catch (e) {
   653→      setNote(e instanceof ApiError && isNotFound(e) ? 'API недоступен' : 'Не удалось записать')
   654→    } finally {
   655→      setBusy(false)
   656→    }
   657→  }
   658→
   659→  return (
   660→    <Panel title={`Вердикт по ${slug}`} icon={<Heart className="size-4" />}>
   661→      <div className="space-y-3">
   662→        <p className="text-xs leading-relaxed text-zinc-500">
   663→          Один клик — и вердикт в логе событий: он кодируется в право (T4-02 уже стал законом 24 слотов).
   664→        </p>
   665→        <div className="flex flex-wrap gap-2">
   666→          {QUICK_VERDICTS.map((v) => (
   667→            <button
   668→              key={v.id}
   669→              onClick={() => setVerdict(verdict === v.id ? null : v.id)}
   670→              className={cn(
   671→                'rounded-md border px-2.5 py-1.5 text-xs font-medium transition-colors',
   672→                verdict === v.id
   673→                  ? 'border-amber-500/60 bg-amber-500/15 text-amber-200'
   674→                  : 'border-zinc-800 bg-zinc-900 text-zinc-400 hover:border-zinc-700 hover:text-zinc-200'
   675→              )}
   676→            >
   677→              {v.label}
   678→            </button>
   679→          ))}
   680→        </div>
   681→        <div className="flex flex-wrap gap-1.5">
   682→          {QUICK_ISSUES.map((x) => (
   683→            <button
   684→              key={x}
   685→              onClick={() => toggleIssue(x)}
   686→              className={cn(
   687→                'rounded-md border px-2 py-1 text-[11px] transition-colors',
   688→                issues.includes(x)
   689→                  ? 'border-rose-500/50 bg-rose-500/10 text-rose-300'
   690→                  : 'border-zinc-800 bg-zinc-900/60 text-zinc-500 hover:border-zinc-700 hover:text-zinc-300'
   691→              )}
   692→            >
   693→              {x}
   694→            </button>
   695→          ))}
   696→        </div>
   697→        <Textarea
   698→          value={text}
   699→          onChange={(e) => setText(e.target.value)}
   700→          placeholder="Детали — как видишь: «на P07 колесо поехало», «Lyn лучше всех»…"
   701→          className="min-h-20 border-zinc-800 bg-zinc-950 text-sm"

   703→        <div className="flex items-center gap-3">
   704→          <Button
   705→            onClick={submit}
   706→            disabled={busy || !verdict}
   707→            className="bg-amber-500 text-zinc-950 hover:bg-amber-400 disabled:opacity-40"

   709→            {busy ? 'Записываю…' : 'Записать вердикт'}
   710→          </Button>
   711→          {note ? <span className="text-xs text-zinc-400">{note}</span> : null}
   712→        </div>

   714→    </Panel>
