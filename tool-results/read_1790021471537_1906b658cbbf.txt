    61→const GENRES = ['OC', 'NICHE', 'VOLT', 'EXQUISITE'] as const
    62→
    63→function genreOf(meta: string): string {
    64→  const u = meta.toUpperCase()
    65→  for (const g of GENRES) {
    66→    if (new RegExp(`\\b${g}\\b`).test(u)) return g
    67→  }
    68→  return ''
    69→}

    71→export function parseBatch(slug: string, text: string): ParsedBatch {
    72→  const titleMatch = /#\s*THREAD 4[^\n]*?Batch\s+(T4-\d+)[^\n]*?["«]([^"»]+)["»]/i.exec(text)
    73→  const title = titleMatch ? titleMatch[2] : ''
    74→  const slots: ParsedSlot[] = []
    75→  const slotRe = /^P(\d{1,2})\s*—\s*(.+)$/gm
    76→  const marks: { pos: number; n: number; anchor: string }[] = []
    77→  let m: RegExpExecArray | null
    78→  while ((m = slotRe.exec(text)) !== null) {
    79→    marks.push({ pos: m.index, n: parseInt(m[1], 10), anchor: m[2] })
    80→  }
    81→  for (let i = 0; i < marks.length; i++) {
    82→    const from = marks[i].pos
    83→    const to = i + 1 < marks.length ? marks[i + 1].pos : text.length
    84→    const block = text.slice(from, to)
    85→    const lines = block.split('\n')
    86→    const header = lines[0] ?? ''
    87→    const grab = (key: string): string => {
    88→      const re = new RegExp(`^${key}:\\s*(.*)$`, 'm')
    89→      const mm = re.exec(block)
    90→      return mm ? mm[1].trim() : ''
    91→    }
    92→    const posMatch = /^POS:\s*\n([\s\S]*?)(?=\nNEG:|\n[^ \n]|\n*$)/m.exec(block)
    93→    const negMatch = /^NEG:\s*\n([\s\S]*?)(?=\n[^ \n]|\n*$)/m.exec(block)
    94→    const stackLine = grab('Stack')
    95→    slots.push({
    96→      position: marks[i].n,
    97→      header,
    98→      anchor: marks[i].anchor.replace(/\s*\(.*$/, ''),
    99→      meta: header.replace(/^[^（(]*[（(]/, '').replace(/[)）]\s*$/, ''),
   100→      genre: genreOf(header.replace(/^[^（(]*[（(]/, '').replace(/[)）]\s*$/, '')),
   101→      thesis: grab('THESIS'),
   102→      canon: grab('Canon'),
   103→      stack: stackLine
   104→        ? (stackLine.match(/CR-[A-Z]\d{2}/g) ?? [])
   105→        : [],
   106→      pos: posMatch ? posMatch[1].trim() : '',
   107→      neg: negMatch ? negMatch[1].trim() : '',
   108→    })
   109→  }
   110→  return { slug, text, title, slots }
   111→}

   117→const TIER_ORDER: Record<string, number> = { 'PG-13': 0, PG13: 0, R: 1, 'R+': 2, RPLUS: 2, X: 3, XXX: 4 }

   119→function tierOf(meta: string): string {
   120→  const u = meta.toUpperCase()
   121→  // order matters: XXX before X, R+ before R («R+ ·» — \b after '+' never fires)
   122→  if (/\bXXX\b/.test(u)) return 'XXX'
   123→  if (/\bR\+/.test(u) || /\bRPLUS\b/.test(u)) return 'R+'
   124→  if (/\bPG-13\b/.test(u) || /\bPG13\b/.test(u)) return 'PG-13'
   125→  if (/\bX\b/.test(u)) return 'X'
   126→  if (/\bR\b/.test(u)) return 'R'
   127→  return ''
   128→}
   129→
   130→function recipeKey(tier: string): string {
   131→  if (tier === 'PG-13') return 'PG13'
   132→  if (tier === 'R+') return 'RPLUS'
   133→  return tier // R, X, XXX
   134→}

   136→function countSignals(haystackLower: string, signals: string[]): number {
   137→  let n = 0
   138→  for (const s of signals) {
   139→    if (haystackLower.includes(s.toLowerCase())) n += 1
   140→  }
   141→  return n
   142→}

   155→export function runGates(slug: string, dryRun = false): GatesResult | null {
   156→  const file = readText(path.join(BATCHES_DIR, `${slug}.md`))
   157→  if (file == null) return null
   158→
   159→  const batch = parseBatch(slug, file)
   160→  const contract = readJson<{
   161→    slots: { position: number; rating: string; race?: string; carriers: { id: string }[] }[]
   162→  }>(path.join(CONTRACTS_DIR, `${slug}.json`))
   163→  const recipes = getRatingRecipes()
   164→  const bans = getBans()
   165→  const carriers = getCarriers()
   166→  const ocCanon = getOCCanon()
   167→
   168→  const receipts: GateReceipt[] = []
   169→  const hard = (gate: string, findings: string[]) =>
   170→    receipts.push({ gate, level: 'hard', verdict: findings.length ? 'FAIL' : 'PASS', findings })
   171→  const warn = (gate: string, findings: string[]) =>
   172→    receipts.push({ gate, level: 'warn', verdict: findings.length ? 'WARN' : 'PASS', findings })
   173→  const advisory = (gate: string, findings: string[]) =>
   174→    receipts.push({ gate, level: 'advisory', verdict: 'REPORT', findings })
   175→
   176→  /* ---------------- 1. structure (hard) ---------------- */
   177→  {
   178→    const f: string[] = []
   179→    if (batch.slots.length !== LAWS.slotsTotal) {
   180→      f.push(`слотов ${batch.slots.length}, ожидается ${LAWS.slotsTotal} (${LAWS.ocSlots} OC + ${LAWS.mainsTotal} мейнов — вердикт T4-02)`)
   181→    }
   182→    const mains = batch.slots.filter((s) => s.genre !== 'OC')
   183→    const ocSlots = batch.slots.filter((s) => s.genre === 'OC')
   184→    if (ocSlots.length !== LAWS.ocSlots) f.push(`OC-слотов ${ocSlots.length}, ожидается ${LAWS.ocSlots}`)
   185→    if (mains.length !== LAWS.mainsTotal) f.push(`мейнов ${mains.length}, ожидается ${LAWS.mainsTotal}`)
   186→    for (const s of batch.slots) {
   187→      if (s.pos === '') f.push(`P${s.position}: пустой POS`)
   188→      if (s.neg === '') f.push(`P${s.position}: пустой NEG`)
   189→      if (s.thesis === '') f.push(`P${s.position}: нет THESIS`)
   190→      if (s.genre === '') f.push(`P${s.position}: жанр не назван в шапке (OC/NICHE/VOLT/EXQUISITE — вердикт T4-02)`)
   191→      if (s.pos && !/masterpiece/i.test(s.pos)) f.push(`P${s.position}: POS без quality-тегов`)
   192→      if (s.pos && !/anime eyes/i.test(s.pos)) f.push(`P${s.position}: нет фейс-лока (anime eyes)`)
   193→      if (s.pos && !/anime style/i.test(s.pos)) f.push(`P${s.position}: нет опенера (anime style)`)
   194→    }
   195→    // OC slots must carry canon lines
   196→    for (const s of batch.slots.filter((x) => x.position <= 3)) {
   197→      if (s.canon === '') f.push(`P${s.position} (OC): нет Canon-строки`)
   198→    }
   199→    hard('structure', f)
   200→  }
   201→
   202→  /* ---------------- 2. floors (hard) ---------------- */
   203→  {
   204→    const f: string[] = []
   205→    const floors = recipes?.eternal_floors ?? {}
   206→    for (const s of batch.slots) {
   207→      const neg = s.neg.toLowerCase()
   208→      for (const term of floors.genital_lock ?? []) {
   209→        if (!neg.includes(term.toLowerCase())) {
   210→          f.push(`P${s.position}: генитальный лок пропускает «${term}»`)
   211→          break
   212→        }
   213→      }
   214→      const loliTerms = floors.anti_loli ?? []
   215→      const loliHits = loliTerms.filter((t) => neg.includes(t.toLowerCase())).length
   216→      if (loliHits < 4) {
   217→        f.push(`P${s.position}: анти-лоли флоор слаб (${loliHits}/${loliTerms.length} терминов) — нужно ≥4`)
   218→      }
   219→      for (const term of floors.leak_guard ?? []) {
   220→        if (!neg.includes(term.toLowerCase())) {
   221→          f.push(`P${s.position}: лика-гарда пропускает «${term}»`)
   222→          break
   223→        }
   224→      }
   225→      // candle guard: required unless the concept itself names the light source
   226→      const posLower = s.pos.toLowerCase()
   227→      const conceptCalls =
   228→        /candle|lantern|torch|chandelier|lamp|brazier/.test(posLower)
   229→      if (!conceptCalls) {
   230→        for (const term of floors.candle_guard ?? []) {
   231→          if (!neg.includes(term.toLowerCase())) {
   232→            f.push(`P${s.position}: свечная гарда пропускает «${term}»`)
   233→            break
   234→          }
   235→        }
   236→      }
   237→      // face guard
   238→      for (const term of floors.face_guard ?? []) {
   239→        if (!neg.includes(term.toLowerCase())) {
   240→          f.push(`P${s.position}: фейс-гарда пропускает «${term}»`)
   241→          break
   242→        }
   243→      }
   244→      // maturity tags (N30): milf is ALWAYS banned in POS; adult woman /
   245→      // mature female are warn-level (batch-wide MILF skew receipt)
   246→      if (/\b(milf|milfs)\b/i.test(s.pos)) {
   247→        f.push(`P${s.position}: тег зрелости в POS («milf») — N30: зрелость только нарративом`)
   248→      }
   249→      // XXX never
   250→      const posLow = s.pos.toLowerCase()
   251→      for (const t of XXX_SIGNALS) {
   252→        if (posLow.includes(t)) {
   253→          f.push(`P${s.position}: XXX-сигнал в POS («${t.trim()}») — вечный потолок`)
   254→          break
   255→        }
   256→      }
   257→    }
   258→    hard('floors', f)
   259→  }
   260→
   261→  /* ---------------- 3. rating-recipe (hard) ---------------- */
   262→  {
   263→    const f: string[] = []
   264→    // EXPLORATORY-слоты (§10: 1-2 слота могут нарушать нежёсткий закон с
   265→    // флагом в THESIS): нарушения RENDER LAW у них — квитанция warn, не
   266→    // hard-fail (эксперимент легален, вердикт автора решает)
   267→    const exploratoryFindings: string[] = []
   268→    const isExploratory = (s: (typeof batch.slots)[number]) => s.header.includes('⚗')
   269→    const tiers = recipes?.tiers ?? {}
   270→    const signalMinDefault: Record<string, number> = { 'PG-13': 2, R: 1, 'R+': 2, X: 2 }
   271→    for (const s of batch.slots) {
   272→      const tier = tierOf(s.meta)
   273→      if (!tier) {
   274→        f.push(`P${s.position}: рейтинг не распознан в шапке («${s.meta}»)`)
   275→        continue
   276→      }
   277→      const recipe = tiers[recipeKey(tier)]
   278→      if (!recipe) continue
   279→      // сигнал-теги считаются в POS — NEG границу держит, а не зарабатывает
   280→      const posLower = s.pos.toLowerCase()
   281→      const negLower = s.neg.toLowerCase()
   282→      const signals = countSignals(posLower, recipe.signals)
   283→      const min = recipe.signal_min ?? signalMinDefault[tier] ?? 1
   284→      if (signals < min) {
   285→        f.push(
   286→          `P${s.position}: заявлен ${tier}, сигнал-тегов рецепта ${signals}/${min} — рейтинг не заработан (рецепт: ${recipe.signals.slice(0, 5).join(', ')}…)`
   287→        )
   288→      }
   289→      // HARD-CLAIM (вердикт T4-03: 14/15 R+ слотов не дотянули): тир с
   290→      // signals_hard обязан нести именованный edge-объект в POS.
   291→      // v1.2.0 (рендер-вердикт T4-03): hard-сигналы = только рендер-
   292→      // доказанные заявки (pantyline / nipples through / see-through / tape);
   293→      // cameltoe выведен — тег рендер-мёртв (0 отрисовок из всех попыток)
   294→      const hardSignals = recipe.signals_hard ?? []
   295→      if (hardSignals.length > 0) {
   296→        const hard = countSignals(posLower, hardSignals)
   297→        const hardMin = recipe.hard_min ?? 1
   298→        if (hard < hardMin) {
   299→          const cameltoeOnly = /\bcameltoe\b|\bcamel toe\b/.test(posLower)
   300→          f.push(
   301→            `P${s.position}: заявлен ${tier}, но рендер-доказанной заявки нет (${hard}/${hardMin})${cameltoeOnly ? ' — cameltoe рендер-мёртв (0 отрисовок, рендер-вердикт T4-03), заявку не зарабатывает' : ''}: ${hardSignals.slice(0, 4).join(', ')}`
   302→          )
   303→        }
   304→      }
   305→      // RENDER LAW v1.2.0 — только для тиров с механизмами (R+): заявка
   306→      // живёт на состоянии ткани, без подслоя, не на мёртвой ткани
   307→      // (присутствие/подслой/мёртвая ткань = hard; позиция тега = warn,
   308→      // см. claim-visibility — рендер-эвиденс позиций неоднозначен)
   309→      const firstDot = s.pos.indexOf('.')
   310→      const tagRun = (firstDot > 0 ? s.pos.slice(0, firstDot) : s.pos).toLowerCase()
   311→      for (const [mechName, m] of Object.entries(recipe.mechanisms ?? {})) {
   312→        const mHard = m.hard ?? []
   313→        const activeHard = mHard.filter((h) => tagRun.includes(h.toLowerCase()))
   314→        if (activeHard.length === 0) continue
   315→        // (а) состояние ткани обязательно в тег-блоке
   316→        const reqState = m.required_state ?? []
   317→        if (reqState.length > 0) {
   318→          const stateHit = reqState.some((st) => tagRun.includes(st.toLowerCase()))
   319→          if (!stateHit) {
   320→            const msg = `P${s.position}: механизм «${mechName}» без состояния ткани — в тег-блоке нет ни одного из [${reqState.join(' / ')}] (рендер-вердикт T4-03: сухая плотная ткань не несёт edge, P13/P21 умерли именно так)`
   321→            if (isExploratory(s)) {
   322→              exploratoryFindings.push(`EXPLORATORY (§10, квитанция): ${msg}`)
   323→            } else {
   324→              f.push(msg)
   325→            }
   326→          }
   327→        }
   328→        // (б) подслой глушит чит: bra/camisole под sheer = рендерер рисует ПОДСЛОЙ
   329→        const blockApplies = m.underlayer_block_applies_to ?? []
   330→        if (blockApplies.some((h) => activeHard.includes(h))) {
   331→          for (const u of m.underlayer_block ?? []) {
   332→            const re = new RegExp(`\\b${u.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\b`)
   333→            if (re.test(tagRun)) {
   334→              f.push(
   335→                `P${s.position}: подслой «${u}» под сквозь-ткань заявкой — рендерер рисует подслой, не edge, тиры падают в R (рендер-вердикт T4-03: P12 sports bra под sheer → R, P14 bra под gauze → R)`
   336→              )
   337→              break
   338→            }
   339→          }
   340→        }
   341→        // (в) мёртвые ткани не несут заявку — зона-осознанно: lower-заявки
   342→        // (pantyline) гибнут на джинсах/коже/свитпанах; upper-заявки
   343→        // (nipples through / see-through) — на бархате/коже/фланели
   344→        const lowerClaims = m.lower_claims ?? []
   345→        const upperClaims = m.upper_claims ?? []
   346→        const isLower = activeHard.some((h) => lowerClaims.includes(h))
   347→        const isUpper = activeHard.some((h) => upperClaims.includes(h))
   348→        const deadList = isLower
   349→          ? [...(m.dead_fabrics_lower ?? []), ...(m.dead_fabrics_upper ?? []).filter((x) => x === 'suit')]
   350→          : isUpper
   351→            ? [...(m.dead_fabrics_upper ?? []), ...(m.dead_fabrics_lower ?? []).filter((x) => x === 'suit')]
   352→            : []
   353→        for (const df of deadList) {
   354→          const re = new RegExp(`\\b${df.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')}\\b`)
   355→          if (re.test(tagRun)) {
   356→            f.push(
   357→              `P${s.position}: мёртвая ткань «${df}» в тег-блоке при активной ${isLower ? 'нижней' : 'грудной'} заявке «${mechName}» — эта ткань не несёт edge (рендер-вердикт T4-03: джинсы/бархат/костюм → PG-13)`
   358→            )
   359→            break
   360→          }
   361→        }
   362→      }
   363→      // механизм-осознанный контр-NEG (вердикт T4-03: «только купальник»):
   364→      // активный механизм держит СВОЙ boundary — lifted-термины в NEG глушат
   365→      for (const [mechName, m] of Object.entries(recipe.mechanisms ?? {})) {
   366→        const mHard = m.hard ?? []
   367→        if (!mHard.some((h) => posLower.includes(h.toLowerCase()))) continue
   368→        for (const t of m.counter_neg ?? []) {
   369→          if (!negLower.includes(t.toLowerCase())) {
   370→            f.push(`P${s.position}: механизм «${mechName}» требует контр-NEG «${t}» в NEG`)
   371→          }
   372→        }
   373→        for (const t of m.counter_neg_lifted ?? []) {
   374→          if (negLower.includes(t.toLowerCase())) {
   375→            f.push(
   376→              `P${s.position}: механизм «${mechName}» заглушен — «${t}» в NEG закрывает грудь и душит сигнал (вердикт T4-03)`
   377→            )
   378→          }
   379→        }
   380→      }
   381→      // counter-NEG: at least one boundary term present
   382→      const counterHit = (recipe.counter_neg ?? []).some((t) => negLower.includes(t.toLowerCase()))
   383→      if (!counterHit) {
   384→        f.push(`P${s.position}: контр-NEG не держит границу тира (нужно ≥1 из: ${(recipe.counter_neg ?? []).slice(0, 3).join(', ')})`)
   385→      }
   386→      // no above-tier signals (overclaim) — but a higher-tier token that is a
   387→      // SUBSTRING of this tier's own signal phrase is by design (e.g. R+ «taped
   388→      // nipples» contains the X token «nipples») — not an overclaim
   389→      const myOrder = TIER_ORDER[tier] ?? 0
   390→      const mySignals = recipe?.signals ?? []
   391→      for (const [otherKey, other] of Object.entries(tiers)) {
   392→        const otherTier =
   393→          otherKey === 'PG13' ? 'PG-13' : otherKey === 'RPLUS' ? 'R+' : otherKey
   394→        if ((TIER_ORDER[otherTier] ?? 0) > myOrder && otherTier !== 'XXX') {
   395→          const over = (other.signals ?? []).filter(
   396→            (sig) =>
   397→              s.pos.toLowerCase().includes(sig.toLowerCase()) &&
   398→              !mySignals.some((m) => m.toLowerCase().includes(sig.toLowerCase()))
   399→          )
   400→          if (over.length >= 1 && tier !== 'X') {
   401→            f.push(`P${s.position}: заявлен ${tier}, но несёт сигнал выше тиром: «${over[0]}»`)
   402→          }
   403→        }
   404→      }
   405→    }
   406→    hard('rating-recipe', f)
   407→    // эксперименты (§10) — квитанцией в claim-visibility (warn), не блоком
   408→    if (exploratoryFindings.length > 0) {
   409→      receipts.push({
   410→        gate: 'rating-recipe-exploratory',
   411→        level: 'warn',
   412→        verdict: 'WARN',
   413→        findings: exploratoryFindings,
   414→      })
   415→    }
   416→  }
   417→
   418→  /* ---------------- 4. canon (hard) ---------------- */
   419→  {
   420→    const f: string[] = []
   421→    const ocText = (name: string) => {

   441→          )

   451→        f.push(`P${s.position}: OC не опознан в шапке`)
   452→        continue
   453→      }
   454→      const oc = ocText(nameMatch[1])

   457→        (w) => w.length > 3 && !['hair', 'with', 'held', 'long', 'into'].includes(w.toLowerCase())

   460→        (w) => w.length > 3 && !['eyes', 'with', 'held'].includes(w.toLowerCase())

   464→      const eyeHit = eyeWords.some((w) => hay.includes(w.toLowerCase()))
   465→      if (!hairHit) f.push(`P${s.position} (${nameMatch[1]}): канон-лок волос не виден`)
   466→      if (!eyeHit) f.push(`P${s.position} (${nameMatch[1]}): канон-лок глаз не виден`)

   779→        const wetIdx = idxs.length > 0 ? Math.min(...idxs) : -1
   780→        const limit = s.genre === 'OC' ? 22 : 16 // 0-based
   781→        if (wetIdx >= limit) {
   782→          f.push(
   783→            `P${s.position}: состояние ткани стоит тегом №${wetIdx + 1} — хвост тег-рана рендерер читает слабо; подними wet clothes / see-through к началу (рендер-вердикт T4-03)`
   784→          )
   785→        }
   786→      }
   787→    }
   788→    warn('claim-visibility', f)
   789→  }
   790→
   791→  /* ---------------- 9. simcheck (warn) ---------------- */
   792→  {
   793→    const f: string[] = []
   794→    const state = foldState(readEvents())
   795→    // бойлерплейт (quality-теги, фейс-лок N31) общий для всех промптов —
   796→    // не считаем предложением автора, иначе J=1.00 на каждом кадре
   797→    const isBoilerplate = (s: string) =>
   798→      /masterpiece|best quality|anime artstyle/.test(s) ||
   799→      /her face is rendered in stylized/.test(s)
   800→    const sentencesOf = (batchParsed: ParsedBatch) =>
   801→      batchParsed.slots.flatMap((s) =>
   802→        s.pos
   803→          .split(/[.!?]\s+/)
   804→          .map((x) => x.trim().toLowerCase())
   805→          .filter((x) => x.length > 40 && !isBoilerplate(x))
   806→      )
   807→    const mySentences = new Set(sentencesOf(batch))
   808→    // повторная сдача (v2): себя не сравниваем — своё окно исключено
   809→    for (const w of state.windowSlugs.filter((x) => x !== batch.slug)) {
   810→      const wText = readText(path.join(BATCHES_DIR, `${w}.md`))
   811→      if (!wText) continue
   812→      const wBatch = parseBatch(w, wText)
   813→      const wSentences = sentencesOf(wBatch)
   814→      for (const mine of mySentences) {
   815→        for (const theirs of wSentences) {
   816→          const j = jaccard(words(mine), words(theirs))
   817→          if (j > 0.7) {
   818→            f.push(`похоже на ${w}: «${mine.slice(0, 60)}…» (J=${j.toFixed(2)})`)
   819→          }
   820→        }
   821→      }
   822→    }
   823→    if (f.length > 8) f.splice(8, f.length, `…и ещё ${f.length - 8} пар`)
   824→    warn('simcheck', f)
   825→  }
   826→
   827→  /* ---------------- 10. freq report (advisory) ---------------- */
   828→  {
   829→    const f: string[] = []
   830→    const STOP = new Set(
   831→      'the a an and or of in on at to with her his its from into over under is are was were be been by as for that this it she they them their not no but all one two'.split(' ')
   832→    )
   833→    const freq: Record<string, number> = {}
   834→    for (const s of batch.slots) {
   835→      const seen = new Set<string>()
   836→      for (const w of s.pos.toLowerCase().match(/[a-z']+/g) ?? []) {
   837→        if (STOP.has(w) || w.length < 5 || seen.has(w)) continue
   838→        seen.add(w)
   839→        freq[w] = (freq[w] ?? 0) + 1
   840→      }
   841→    }
   842→    const tics = Object.entries(freq)
   843→      .filter(([, n]) => n >= 8)
   844→      .sort((a, b) => b[1] - a[1])
   845→      .slice(0, 12)
   846→      .map(([w, n]) => `${w} ${n}/21`)
   847→    if (tics.length) f.push(`кандидаты в тики: ${tics.join(' · ')}`)
   848→    else f.push('тиков не обнаружено')
   849→    advisory('freq-report', f)
   850→  }
   851→
   852→  /* ---------------- 11. usage (advisory) ---------------- */
   853→  {
   854→    const f: string[] = []
   855→    const stacked = batch.slots.flatMap((s) => s.stack)
   856→    f.push(`носителей в стеках: ${stacked.length}; уникальных: ${new Set(stacked).size}`)
   857→    f.push(`R+ слотов: ${batch.slots.filter((s) => tierOf(s.meta) === 'R+').length}; X: ${batch.slots.filter((s) => tierOf(s.meta) === 'X').length}`)
   858→    advisory('usage', f)
   859→  }
   860→
   861→  /* ---------------- receipt + event ---------------- */
   862→  const sha10 = crypto.createHash('sha256').update(file).digest('hex').slice(0, 10)
   863→  const priorRuns = readEvents().filter(
   864→    (e) => e.type === 'gate.run' && e.data?.slug === slug
   865→  ).length
   866→  const runIndex = priorRuns + 1
   867→  const hardPass = receipts.filter((r) => r.level === 'hard').every((r) => r.verdict === 'PASS')
   868→  const result: GatesResult = {
   869→    slug,
   870→    runIndex,
   871→    sha10,
   872→    receipts,
   873→    hardPass,
   874→    firstRunClean: hardPass && runIndex === 1,
   875→    at: new Date().toISOString(),
   876→  }
   877→  if (!dryRun) {
   878→    appendEvent(
   879→      'gate.run',
   880→      `${slug} гейты: прогон #${runIndex}, hard ${hardPass ? 'PASS' : 'FAIL'}${result.firstRunClean ? ' · FIRST RUN CLEAN' : ''} (sha ${sha10})`,
   881→      {
   882→        slug,
   883→        run: runIndex,
   884→        sha10,
   885→        hardPass,
   886→        firstRunClean: result.firstRunClean,
   887→        hardFails: receipts
   888→          .filter((r) => r.level === 'hard' && r.verdict === 'FAIL')
   889→          .map((r) => r.gate),
   890→        warns: receipts.filter((r) => r.level === 'warn' && r.verdict === 'WARN').length,
   891→      }
   892→    )
   893→  }
   894→  return result
   895→}

   901→function words(s: string): Set<string> {
   902→  return new Set((s.toLowerCase().match(/[a-z']+/g) ?? []).filter((w) => w.length > 3))
   903→}
   904→
   905→function jaccard(a: Set<string>, b: Set<string>): number {
   906→  if (a.size === 0 || b.size === 0) return 0
   907→  let inter = 0
   908→  for (const x of a) if (b.has(x)) inter += 1
   909→  return inter / (a.size + b.size - inter)
   910→}
